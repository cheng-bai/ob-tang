    {\LARGE\bfseries 专题：数列的通项与求和}
\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

数列的通项公式与求和公式是数列研究的核心内容。对于非等差、非等比的一般数列，需要通过转化、构造等方法求解。

\begin{definition}[数列的通项公式]
    如果数列 $\{a_n\}$ 的第 $n$ 项 $a_n$ 与序号 $n$ 之间的关系可以用一个公式表示，那么这个公式叫做数列的\textbf{通项公式}。
\end{definition}

\begin{definition}[数列的前 $n$ 项和]
    设数列 $\{a_n\}$ 的前 $n$ 项和为 $S_n$，即
    \[
        S_n = a_1 + a_2 + \cdots + a_n = \sum_{k=1}^{n} a_k.
    \]
    通项与前 $n$ 项和的关系：
    \[
        a_n = \begin{cases}
            S_1, & n = 1, \\[6pt]
            S_n - S_{n-1}, & n \geq 2.
        \end{cases}
    \]
\end{definition}

\begin{property}[由递推公式求通项的常见类型]
    \begin{enumerate}
        \item \textbf{累加法}：$a_{n+1} = a_n + f(n)$ 型
        \[
            a_n = a_1 + \sum_{k=1}^{n-1} f(k) \quad (n \geq 2).
        \]
        
        \item \textbf{累乘法}：$a_{n+1} = a_n \cdot f(n)$ 型
        \[
            a_n = a_1 \cdot \prod_{k=1}^{n-1} f(k) \quad (n \geq 2).
        \]
        
        \item \textbf{构造等比法}：$a_{n+1} = pa_n + q$（$p \neq 1, q \neq 0$）型
        \[
            a_{n+1} + \frac{q}{p-1} = p\left(a_n + \frac{q}{p-1}\right).
        \]
        数列 $\left\{a_n + \dfrac{q}{p-1}\right\}$ 是公比为 $p$ 的等比数列。
        
        \item \textbf{构造等差法}：$a_{n+1} - a_n = f(n) \cdot a_{n+1} \cdot a_n$ 型，两边除以 $a_{n+1} \cdot a_n$。
    \end{enumerate}
\end{property}

\begin{property}[数列求和的常用方法]
    \begin{enumerate}
        \item \textbf{公式法}：直接利用等差、等比数列求和公式。
        
        \item \textbf{分组求和法}：将数列分解为几个等差、等比或可求和数列的和。
        
        \item \textbf{错位相减法}：适用于 $\{a_n \cdot b_n\}$ 型，其中 $\{a_n\}$ 是等差数列，$\{b_n\}$ 是等比数列。
        
        \item \textbf{裂项相消法}：将通项拆成两项之差，求和时中间项相互抵消。
        \begin{itemize}
            \item $\dfrac{1}{n(n+1)} = \dfrac{1}{n} - \dfrac{1}{n+1}$
            \item $\dfrac{1}{n(n+k)} = \dfrac{1}{k}\left(\dfrac{1}{n} - \dfrac{1}{n+k}\right)$
            \item $\dfrac{1}{\sqrt{n} + \sqrt{n+1}} = \sqrt{n+1} - \sqrt{n}$
        \end{itemize}
        
        \item \textbf{倒序相加法}：适用于首尾对称的数列，如等差数列求和。
        
        \item \textbf{并项求和法}：将相邻若干项合并为一组求和。
    \end{enumerate}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[由 $S_n$ 求 $a_n$]
    已知数列 $\{a_n\}$ 的前 $n$ 项和 $S_n = n^2 + 2n$，求数列 $\{a_n\}$ 的通项公式。
\end{example}

\textbf{解.} 当 $n = 1$ 时，$a_1 = S_1 = 1^2 + 2 \times 1 = 3$。

当 $n \geq 2$ 时：
\[
    a_n = S_n - S_{n-1} = (n^2 + 2n) - [(n-1)^2 + 2(n-1)]
\]
\[
    = n^2 + 2n - (n^2 - 2n + 1 + 2n - 2) = n^2 + 2n - n^2 + 1 = 2n + 1.
\]

验证：当 $n = 1$ 时，$2 \times 1 + 1 = 3 = a_1$，符合上式。

因此，$a_n = 2n + 1$（$n \in \mathbb{N}^*$）。

\begin{example}[累加法求通项]
    已知数列 $\{a_n\}$ 满足 $a_1 = 12$，$a_{n+1} = a_n + 2n$（$n \geq 1$），求 $\dfrac{a_n}{n}$ 的最小值。
\end{example}

\textbf{解.} 由递推公式 $a_{n+1} - a_n = 2n$，利用累加法：

当 $n \geq 2$ 时：
\[
    a_n = a_1 + \sum_{k=1}^{n-1} 2k = 12 + 2 \cdot \frac{(n-1)n}{2} = 12 + n(n-1) = n^2 - n + 12.
\]

验证 $n = 1$：$a_1 = 1 - 1 + 12 = 12$，符合。

所以 $a_n = n^2 - n + 12$，则
\[
    \frac{a_n}{n} = \frac{n^2 - n + 12}{n} = n - 1 + \frac{12}{n}.
\]

设 $f(n) = n + \dfrac{12}{n} - 1$，求其最小值。

由均值不等式，$n + \dfrac{12}{n} \geq 2\sqrt{12} = 4\sqrt{3} \approx 6.93$，当且仅当 $n = \sqrt{12} \approx 3.46$ 时取等。

由于 $n \in \mathbb{N}^*$，计算：
\begin{align*}
    f(3) &= 3 + \frac{12}{3} - 1 = 3 + 4 - 1 = 6, \\
    f(4) &= 4 + \frac{12}{4} - 1 = 4 + 3 - 1 = 6.
\end{align*}

因此，$\dfrac{a_n}{n}$ 的最小值为 $6$。

\begin{realexam}[2023 普陀一模改编]
    若数列 $\{a_n\}$ 满足 $a_1 = 12$，$a_{n+1} = a_n + 2n$（$n \geq 1$），则 $\dfrac{a_n}{n}$ 的最小值是\_\_\_。
\end{realexam}

\textbf{解.} 同上，答案为 $6$。

\begin{example}[构造等比数列求通项]
    已知各项均为正数的数列 $\{a_n\}$ 满足 $a_1 = 1$，$a_n = 2a_{n-1} + 3$（正整数 $n \geq 2$）。
    \begin{enumerate}
        \item 求证：数列 $\{a_n + 3\}$ 是等比数列；
        \item 求数列 $\{a_n\}$ 的前 $n$ 项和 $S_n$。
    \end{enumerate}
\end{example}

\textbf{解.} (1) 由 $a_n = 2a_{n-1} + 3$，两边同时加 $3$：
\[
    a_n + 3 = 2a_{n-1} + 6 = 2(a_{n-1} + 3).
\]

所以 $\dfrac{a_n + 3}{a_{n-1} + 3} = 2$（常数）。

又 $a_1 + 3 = 4$，故数列 $\{a_n + 3\}$ 是以 $4$ 为首项、以 $2$ 为公比的等比数列。

(2) 由 (1) 得：
\[
    a_n + 3 = 4 \cdot 2^{n-1} = 2^{n+1},
\]
所以 $a_n = 2^{n+1} - 3$。

前 $n$ 项和：
\[
    S_n = \sum_{k=1}^{n} (2^{k+1} - 3) = \sum_{k=1}^{n} 2^{k+1} - 3n = 4 \cdot \frac{2^n - 1}{2-1} - 3n = 2^{n+2} - 4 - 3n.
\]

\begin{example}[裂项相消法求和]
    已知数列 $\{a_n\}$ 的通项公式为 $a_n = \dfrac{1}{n(n+1)}$，求该数列的前 $n$ 项和 $S_n$。
\end{example}

\textbf{解.} 利用裂项相消法：
\[
    a_n = \frac{1}{n(n+1)} = \frac{1}{n} - \frac{1}{n+1}.
\]

前 $n$ 项和：
\[
    S_n = \sum_{k=1}^{n} \left(\frac{1}{k} - \frac{1}{k+1}\right) = \left(1 - \frac{1}{2}\right) + \left(\frac{1}{2} - \frac{1}{3}\right) + \cdots + \left(\frac{1}{n} - \frac{1}{n+1}\right).
\]

中间项相互抵消，得：
\[
    S_n = 1 - \frac{1}{n+1} = \frac{n}{n+1}.
\]

\begin{example}[错位相减法求和]
    求数列 $\{n \cdot 2^n\}$ 的前 $n$ 项和 $S_n$。
\end{example}

\textbf{解.} 设 $S_n = 1 \cdot 2^1 + 2 \cdot 2^2 + 3 \cdot 2^3 + \cdots + n \cdot 2^n$。

两边乘以公比 $2$：
\[
    2S_n = 1 \cdot 2^2 + 2 \cdot 2^3 + \cdots + (n-1) \cdot 2^n + n \cdot 2^{n+1}.
\]

两式相减：
\[
    2S_n - S_n = -2^1 - 2^2 - 2^3 - \cdots - 2^n + n \cdot 2^{n+1}.
\]

即：
\[
    S_n = n \cdot 2^{n+1} - (2 + 2^2 + \cdots + 2^n) = n \cdot 2^{n+1} - \frac{2(2^n - 1)}{2-1} = n \cdot 2^{n+1} - 2^{n+1} + 2 = (n-1) \cdot 2^{n+1} + 2.
\]

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{由 $S_n$ 求 $a_n$}：务必分 $n = 1$ 和 $n \geq 2$ 讨论，最后验证 $n = 1$ 是否满足通式。
    
    \item \textbf{递推公式求通项}：
    \begin{itemize}
        \item $a_{n+1} = a_n + f(n)$ $\Rightarrow$ 累加法
        \item $a_{n+1} = f(n) \cdot a_n$ $\Rightarrow$ 累乘法
        \item $a_{n+1} = pa_n + q$ $\Rightarrow$ 构造等比数列
        \item $a_{n+1} = pa_n + q^n$ $\Rightarrow$ 两边除以 $q^{n+1}$ 或构造
    \end{itemize}
    
    \item \textbf{数列求和方法选择}：
    \begin{itemize}
        \item 等差、等比数列 $\Rightarrow$ 公式法
        \item $\{a_n \pm b_n\}$ 型 $\Rightarrow$ 分组求和法
        \item $\{a_n \cdot b_n\}$ 型（等差$\times$等比）$\Rightarrow$ 错位相减法
        \item 分式型、根式型 $\Rightarrow$ 裂项相消法
        \item 首尾对称型 $\Rightarrow$ 倒序相加法
        \item 摆动数列 $\Rightarrow$ 并项求和法
    \end{itemize}
    
    \item \textbf{裂项技巧}：
    \begin{itemize}
        \item $\dfrac{1}{n(n+k)} = \dfrac{1}{k}\left(\dfrac{1}{n} - \dfrac{1}{n+k}\right)$
        \item $\dfrac{1}{\sqrt{n} + \sqrt{n+k}} = \dfrac{1}{k}(\sqrt{n+k} - \sqrt{n})$
        \item $\dfrac{2^n}{(2^n+1)(2^{n+1}+1)} = \dfrac{1}{2^n+1} - \dfrac{1}{2^{n+1}+1}$
    \end{itemize}
    
    \item \textbf{错位相减法要点}：
    \begin{itemize}
        \item 写出 $S_n$ 的表达式
        \item 两边乘以等比数列的公比 $q$
        \item 两式相减（注意对齐项）
        \item 化简求和
    \end{itemize}
\end{enumerate}
