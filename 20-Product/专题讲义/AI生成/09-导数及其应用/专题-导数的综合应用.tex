\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

导数作为研究函数性质的有力工具，在解决方程、不等式、函数零点等问题中有广泛应用。

\begin{property}[利用导数研究函数零点]
    函数 $f(x)$ 的零点即方程 $f(x) = 0$ 的解。利用导数可以：
    \begin{itemize}
        \item 通过单调性判断零点个数：单调函数最多有一个零点；
        \item 通过极值判断零点个数：若极大值小于零或极小值大于零，则无零点；
        \item 利用零点存在定理结合单调性确定零点所在区间。
    \end{itemize}
\end{property}

\begin{property}[利用导数证明不等式]
    证明 $f(x) > g(x)$ 的常用方法：
    \begin{itemize}
        \item 构造函数 $h(x) = f(x) - g(x)$，证明 $h(x) > 0$；
        \item 利用单调性：证明 $h(x)$ 单调递增且 $h(a) \ge 0$；
        \item 利用最值：证明 $h(x)$ 的最小值大于零；
        \item 利用凹凸性：结合 Jensen 不等式等。
    \end{itemize}
\end{property}

\begin{property}[恒成立与存在性问题]
    \begin{itemize}
        \item $f(x) > 0$ 在区间 $I$ 上恒成立 $\Leftrightarrow$ $f(x)$ 在 $I$ 上的最小值 $> 0$；
        \item $f(x) > 0$ 在区间 $I$ 上有解（存在性）$\Leftrightarrow$ $f(x)$ 在 $I$ 上的最大值 $> 0$；
        \item $f(x) > g(x)$ 恒成立 $\Leftrightarrow$ $[f(x) - g(x)]_{\min} > 0$；
        \item $f(x) > g(x)$ 有解 $\Leftrightarrow$ $[f(x) - g(x)]_{\max} > 0$。
    \end{itemize}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[函数零点个数问题]
    讨论函数 $f(x) = \ln x - ax$（$a > 0$）的零点个数。
\end{example}

\textbf{解.} 定义域为 $(0, +\infty)$。

求导：$f'(x) = \dfrac{1}{x} - a = \dfrac{1 - ax}{x}$。

令 $f'(x) = 0$，得 $x = \dfrac{1}{a}$。

当 $0 < x < \dfrac{1}{a}$ 时，$f'(x) > 0$，$f(x)$ 单调递增；

当 $x > \dfrac{1}{a}$ 时，$f'(x) < 0$，$f(x)$ 单调递减。

所以 $x = \dfrac{1}{a}$ 是极大值点，极大值为：
\[
    f\left(\frac{1}{a}\right) = \ln\frac{1}{a} - a \cdot \frac{1}{a} = -\ln a - 1.
\]

分析极限：
\begin{itemize}
    \item 当 $x \to 0^+$ 时，$\ln x \to -\infty$，所以 $f(x) \to -\infty$；
    \item 当 $x \to +\infty$ 时，$\ln x$ 增长慢于 $ax$，所以 $f(x) \to -\infty$。
\end{itemize}

\textbf{情况1}：当极大值 $-\ln a - 1 < 0$，即 $a > \dfrac{1}{e}$ 时，$f(x) < 0$ 恒成立，无零点。

\textbf{情况2}：当极大值 $-\ln a - 1 = 0$，即 $a = \dfrac{1}{e}$ 时，$f(x) \le 0$ 且仅在 $x = e$ 处取等号，有一个零点。

\textbf{情况3}：当极大值 $-\ln a - 1 > 0$，即 $0 < a < \dfrac{1}{e}$ 时，由零点存在定理和单调性，$f(x)$ 在 $\left(0, \dfrac{1}{a}\right)$ 和 $\left(\dfrac{1}{a}, +\infty\right)$ 上各有一个零点，共两个零点。

\begin{example}[不等式恒成立问题]
    已知函数 $f(x) = e^x - ax - 1$，若 $f(x) \ge 0$ 对所有 $x \in \mathbb{R}$ 恒成立，求实数 $a$ 的取值范围。
\end{example}

\textbf{解.} 求导：$f'(x) = e^x - a$。

\textbf{情况1}：当 $a \le 0$ 时，$f'(x) = e^x - a > 0$ 恒成立，$f(x)$ 在 $\mathbb{R}$ 上单调递增。

当 $x \to -\infty$ 时，$f(x) \to -\infty$，不满足 $f(x) \ge 0$ 恒成立。

\textbf{情况2}：当 $a > 0$ 时，令 $f'(x) = 0$，得 $x = \ln a$。

当 $x < \ln a$ 时，$f'(x) < 0$，$f(x)$ 单调递减；

当 $x > \ln a$ 时，$f'(x) > 0$，$f(x)$ 单调递增。

所以 $x = \ln a$ 是最小值点，最小值为：
\[
    f(\ln a) = e^{\ln a} - a \cdot \ln a - 1 = a - a\ln a - 1 = a(1 - \ln a) - 1.
\]

要使 $f(x) \ge 0$ 恒成立，需 $f(\ln a) \ge 0$，即
\[
    a - a\ln a - 1 \ge 0.
\]

设 $g(a) = a - a\ln a - 1$（$a > 0$），则 $g'(a) = 1 - \ln a - 1 = -\ln a$。

当 $0 < a < 1$ 时，$g'(a) > 0$，$g(a)$ 单调递增；

当 $a > 1$ 时，$g'(a) < 0$，$g(a)$ 单调递减。

所以 $g(a)$ 在 $a = 1$ 处取得最大值 $g(1) = 1 - 0 - 1 = 0$。

因此 $g(a) \le 0$ 恒成立，且仅当 $a = 1$ 时 $g(a) = 0$。

综上，$a = 1$。

\begin{realexam}[2024 上海一模]
    已知函数 $f(x) = x^3 - 3x + a$ 有三个不同的零点，求实数 $a$ 的取值范围。
\end{realexam}

\textbf{解.} 求导：$f'(x) = 3x^2 - 3 = 3(x + 1)(x - 1)$。

令 $f'(x) = 0$，得 $x_1 = -1$，$x_2 = 1$。

当 $x < -1$ 或 $x > 1$ 时，$f'(x) > 0$，$f(x)$ 单调递增；

当 $-1 < x < 1$ 时，$f'(x) < 0$，$f(x)$ 单调递减。

所以 $x = -1$ 是极大值点，$x = 1$ 是极小值点。

计算极值：
\begin{align*}
    f(-1) &= (-1)^3 - 3(-1) + a = -1 + 3 + a = a + 2, \\[4pt]
    f(1) &= 1 - 3 + a = a - 2.
\end{align*}

要使 $f(x)$ 有三个不同的零点，需极大值大于零且极小值小于零：
\[
    \begin{cases}
        a + 2 > 0 \\[4pt]
        a - 2 < 0
    \end{cases}
\]

解得 $-2 < a < 2$。

\begin{realexam}[2025 上海二模]
    证明：当 $x > 0$ 时，$\ln(x + 1) > \dfrac{x}{x + 1}$。
\end{realexam}

\textbf{解.} 构造函数 $f(x) = \ln(x + 1) - \dfrac{x}{x + 1}$，定义域为 $[0, +\infty)$。

求导：
\[
    f'(x) = \frac{1}{x + 1} - \frac{(x + 1) - x}{(x + 1)^2} = \frac{1}{x + 1} - \frac{1}{(x + 1)^2} = \frac{x}{(x + 1)^2}.
\]

当 $x > 0$ 时，$f'(x) = \dfrac{x}{(x + 1)^2} > 0$，所以 $f(x)$ 在 $[0, +\infty)$ 上单调递增。

又 $f(0) = \ln 1 - 0 = 0$，所以当 $x > 0$ 时，$f(x) > f(0) = 0$。

即 $\ln(x + 1) - \dfrac{x}{x + 1} > 0$，所以 $\ln(x + 1) > \dfrac{x}{x + 1}$。证毕。

\begin{example}[双变量问题]
    已知函数 $f(x) = x - \ln x$，若对任意 $x_1, x_2 \in [1, e]$，都有 $|f(x_1) - f(x_2)| \le a$，求实数 $a$ 的最小值。
\end{example}

\textbf{解.} 问题等价于求 $f(x)$ 在 $[1, e]$ 上的最大值与最小值之差。

求导：$f'(x) = 1 - \dfrac{1}{x} = \dfrac{x - 1}{x}$。

当 $x \in [1, e]$ 时，$f'(x) \ge 0$，所以 $f(x)$ 在 $[1, e]$ 上单调递增。

因此：
\begin{align*}
    f(x)_{\min} &= f(1) = 1 - \ln 1 = 1, \\[4pt]
    f(x)_{\max} &= f(e) = e - \ln e = e - 1.
\end{align*}

所以 $|f(x_1) - f(x_2)|_{\max} = f(e) - f(1) = (e - 1) - 1 = e - 2$。

要使 $|f(x_1) - f(x_2)| \le a$ 恒成立，需 $a \ge e - 2$。

故 $a$ 的最小值为 $e - 2$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{零点问题策略}：先利用导数分析函数的单调性和极值，结合极限分析函数图像的大致形状，再根据极值与零的关系判断零点个数。
    \item \textbf{不等式证明方法}：构造函数法是核心，通过研究新函数的单调性或最值来证明不等式。有时需要对不等式进行等价变形。
    \item \textbf{恒成立问题转化}：恒成立问题通常转化为求函数最值问题。关键是正确理解"恒成立"与"存在性"的区别。
    \item \textbf{参数讨论技巧}：含参问题要找准分类标准，通常从导数的符号、极值点的位置、极值的正负等方面进行分类。
    \item \textbf{端点效应}：对于区间端点处取等号的不等式恒成立问题，可以利用端点处的导数信息缩小参数范围。
    \item \textbf{放缩技巧}：证明不等式时，适当的放缩可以简化问题。常用放缩如 $e^x \ge x + 1$，$\ln x \le x - 1$ 等。
\end{enumerate}
