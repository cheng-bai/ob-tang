\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

函数的极值与最值是导数应用的重要内容，在实际问题中有广泛应用。

\begin{definition}[极值]
    设函数 $f(x)$ 在点 $x_0$ 的某邻域内有定义，若对该邻域内任意 $x \neq x_0$，都有
    \begin{itemize}
        \item $f(x) < f(x_0)$，则称 $f(x_0)$ 为函数的\textbf{极大值}，$x_0$ 为\textbf{极大值点}；
        \item $f(x) > f(x_0)$，则称 $f(x_0)$ 为函数的\textbf{极小值}，$x_0$ 为\textbf{极小值点}。
    \end{itemize}
    极大值与极小值统称为\textbf{极值}，极大值点与极小值点统称为\textbf{极值点}。
\end{definition}

\begin{property}[极值的必要条件]
    若函数 $f(x)$ 在点 $x_0$ 处可导且取得极值，则 $f'(x_0) = 0$。
    
    使 $f'(x) = 0$ 的点称为函数的\textbf{驻点}（或稳定点）。
    
    \textbf{注意}：可导函数的极值点一定是驻点，但驻点不一定是极值点。
\end{property}

\begin{property}[极值的第一充分条件]
    设函数 $f(x)$ 在点 $x_0$ 处连续，在 $x_0$ 的某去心邻域内可导：
    \begin{itemize}
        \item 若 $f'(x)$ 在 $x_0$ 左侧为负、右侧为正，则 $x_0$ 是极小值点；
        \item 若 $f'(x)$ 在 $x_0$ 左侧为正、右侧为负，则 $x_0$ 是极大值点；
        \item 若 $f'(x)$ 在 $x_0$ 两侧同号，则 $x_0$ 不是极值点。
    \end{itemize}
\end{property}

\begin{property}[极值的第二充分条件]
    设函数 $f(x)$ 在点 $x_0$ 处二阶可导，且 $f'(x_0) = 0$：
    \begin{itemize}
        \item 若 $f''(x_0) > 0$，则 $x_0$ 是极小值点；
        \item 若 $f''(x_0) < 0$，则 $x_0$ 是极大值点；
        \item 若 $f''(x_0) = 0$，则无法判断，需用第一充分条件。
    \end{itemize}
\end{property}

\begin{property}[闭区间上最值的求法]
    设函数 $f(x)$ 在闭区间 $[a, b]$ 上连续，求最值的步骤：
    \begin{enumerate}
        \item 求 $f(x)$ 在 $(a, b)$ 内的所有驻点和不可导点；
        \item 计算 $f(x)$ 在这些点及区间端点 $a, b$ 处的函数值；
        \item 比较上述函数值，最大者为最大值，最小者为最小值。
    \end{enumerate}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[求函数的极值]
    求函数 $f(x) = x^3 - 3x^2 - 9x + 5$ 的极值。
\end{example}

\textbf{解.} 定义域为 $\mathbb{R}$。

求导：$f'(x) = 3x^2 - 6x - 9 = 3(x + 1)(x - 3)$。

令 $f'(x) = 0$，得驻点 $x_1 = -1$，$x_2 = 3$。

列表分析：
\begin{center}
\begin{tabular}{c|ccccc}
\hline
$x$ & $(-\infty, -1)$ & $-1$ & $(-1, 3)$ & $3$ & $(3, +\infty)$ \\
\hline
$f'(x)$ & $+$ & $0$ & $-$ & $0$ & $+$ \\
\hline
$f(x)$ & $\nearrow$ & 极大值 & $\searrow$ & 极小值 & $\nearrow$ \\
\hline
\end{tabular}
\end{center}

计算极值：
\begin{align*}
    f(-1) &= (-1)^3 - 3(-1)^2 - 9(-1) + 5 = -1 - 3 + 9 + 5 = 10, \\[4pt]
    f(3) &= 27 - 27 - 27 + 5 = -22.
\end{align*}

因此，$f(x)$ 的极大值为 $10$（在 $x = -1$ 处取得），极小值为 $-22$（在 $x = 3$ 处取得）。

\begin{example}[利用二阶导数判断极值]
    求函数 $f(x) = x e^{-x}$ 的极值。
\end{example}

\textbf{解.} 定义域为 $\mathbb{R}$。

求导：$f'(x) = e^{-x} + x \cdot (-e^{-x}) = e^{-x}(1 - x)$。

令 $f'(x) = 0$，由于 $e^{-x} > 0$，所以 $1 - x = 0$，得驻点 $x = 1$。

求二阶导数：
\[
    f''(x) = -e^{-x}(1 - x) + e^{-x}(-1) = e^{-x}(x - 2).
\]

在 $x = 1$ 处：$f''(1) = e^{-1}(1 - 2) = -\dfrac{1}{e} < 0$。

由第二充分条件，$x = 1$ 是极大值点。

极大值为 $f(1) = 1 \cdot e^{-1} = \dfrac{1}{e}$。

\begin{example}[求闭区间上的最值]
    求函数 $f(x) = x^3 - 3x + 1$ 在区间 $[-2, 2]$ 上的最大值和最小值。
\end{example}

\textbf{解.} 求导：$f'(x) = 3x^2 - 3 = 3(x + 1)(x - 1)$。

令 $f'(x) = 0$，得驻点 $x_1 = -1$，$x_2 = 1$，均在区间 $[-2, 2]$ 内。

计算各关键点函数值：
\begin{align*}
    f(-2) &= (-2)^3 - 3(-2) + 1 = -8 + 6 + 1 = -1, \\[4pt]
    f(-1) &= (-1)^3 - 3(-1) + 1 = -1 + 3 + 1 = 3, \\[4pt]
    f(1) &= 1 - 3 + 1 = -1, \\[4pt]
    f(2) &= 8 - 6 + 1 = 3.
\end{align*}

比较得：最大值为 $3$（在 $x = -1$ 和 $x = 2$ 处取得），最小值为 $-1$（在 $x = -2$ 和 $x = 1$ 处取得）。

\begin{realexam}[2024 上海一模]
    已知函数 $f(x) = \dfrac{1}{3}x^3 - \dfrac{1}{2}x^2 - 2x + 1$，求 $f(x)$ 在区间 $[0, 3]$ 上的最大值和最小值。
\end{realexam}

\textbf{解.} 求导：$f'(x) = x^2 - x - 2 = (x - 2)(x + 1)$。

令 $f'(x) = 0$，得 $x_1 = -1$（不在区间内），$x_2 = 2$（在区间内）。

计算关键点函数值：
\begin{align*}
    f(0) &= 1, \\[4pt]
    f(2) &= \frac{8}{3} - 2 - 4 + 1 = \frac{8}{3} - 5 = -\frac{7}{3}, \\[4pt]
    f(3) &= 9 - \frac{9}{2} - 6 + 1 = \frac{18 - 9 - 12 + 2}{2} = -\frac{1}{2}.
\end{align*}

比较得：最大值为 $1$（在 $x = 0$ 处取得），最小值为 $-\dfrac{7}{3}$（在 $x = 2$ 处取得）。

\begin{realexam}[2023 上海秋考]
    某工厂生产一种产品，已知固定成本为 $20000$ 元，每生产一件产品，成本增加 $100$ 元。该产品的需求函数为 $Q = 500 - 2P$（$Q$ 为需求量，$P$ 为价格）。求使利润最大时的产量和价格。
\end{realexam}

\textbf{解.} 由 $Q = 500 - 2P$，得 $P = 250 - \dfrac{Q}{2}$。

收入函数：$R(Q) = P \cdot Q = \left(250 - \dfrac{Q}{2}\right)Q = 250Q - \dfrac{Q^2}{2}$。

成本函数：$C(Q) = 20000 + 100Q$。

利润函数：
\[
    L(Q) = R(Q) - C(Q) = 250Q - \frac{Q^2}{2} - 20000 - 100Q = 150Q - \frac{Q^2}{2} - 20000.
\]

求导：$L'(Q) = 150 - Q$。

令 $L'(Q) = 0$，得 $Q = 150$。

当 $0 < Q < 150$ 时，$L'(Q) > 0$；当 $Q > 150$ 时，$L'(Q) < 0$。

所以 $Q = 150$ 是极大值点，也是最大值点。

此时价格 $P = 250 - \dfrac{150}{2} = 250 - 75 = 175$（元）。

答：使利润最大时的产量为 $150$ 件，价格为 $175$ 元。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{极值与最值的区别}：极值是局部概念，是某邻域内的最大或最小值；最值是整体概念，是定义域内的最大或最小值。极值点不一定是最值点，最值点也不一定是极值点（可能在端点处取得）。
    \item \textbf{求极值的步骤}：先求导找驻点，再用第一或第二充分条件判断是否为极值点，最后计算极值。
    \item \textbf{第二充分条件的适用}：当 $f''(x_0)$ 易求且不为零时，用第二充分条件更简便；若 $f''(x_0) = 0$ 或难求，则用第一充分条件。
    \item \textbf{闭区间最值}：必须在驻点、不可导点和端点处都计算函数值，然后比较大小。
    \item \textbf{实际应用问题}：建立目标函数时，注意确定自变量的取值范围（定义域），并验证结果的实际意义。
    \item \textbf{注意陷阱}：导数不存在的点也可能是极值点（如 $f(x) = |x|$ 在 $x = 0$ 处），不要遗漏。
\end{enumerate}
