\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

抛物线是圆锥曲线的重要类型，其定义和性质在解析几何中具有独特地位。掌握抛物线的定义、标准方程和几何性质是解题的基础。

\begin{definition}[抛物线的定义]
    平面内与一个定点 $F$（称为焦点）和一条定直线 $l$（称为准线，$F \notin l$）距离相等的点的轨迹叫做抛物线。即
    \[
        |PF| = d(P, l)
    \]
    其中 $d(P, l)$ 表示点 $P$ 到直线 $l$ 的距离。
\end{definition}

\begin{property}[标准方程与几何性质]
    \textbf{标准方程（四种形式）：}
    \begin{center}
    \begin{tabular}{|c|c|c|c|c|}
        \hline
        方程 & 焦点 & 准线 & 开口方向 & 焦准距 \\
        \hline
        $y^2 = 2px$（$p > 0$） & $\left(\dfrac{p}{2}, 0\right)$ & $x = -\dfrac{p}{2}$ & 向右 & $p$ \\
        \hline
        $y^2 = -2px$（$p > 0$） & $\left(-\dfrac{p}{2}, 0\right)$ & $x = \dfrac{p}{2}$ & 向左 & $p$ \\
        \hline
        $x^2 = 2py$（$p > 0$） & $\left(0, \dfrac{p}{2}\right)$ & $y = -\dfrac{p}{2}$ & 向上 & $p$ \\
        \hline
        $x^2 = -2py$（$p > 0$） & $\left(0, -\dfrac{p}{2}\right)$ & $y = \dfrac{p}{2}$ & 向下 & $p$ \\
        \hline
    \end{tabular}
    \end{center}
    
    \textbf{注：}焦准距（焦点到准线的距离）为 $p$，是抛物线的重要参数。
\end{property}

\begin{property}[焦点弦性质]
    设抛物线 $y^2 = 2px$（$p > 0$）的焦点为 $F$，过 $F$ 的直线交抛物线于 $A(x_1, y_1)$、$B(x_2, y_2)$ 两点，则：
    \begin{itemize}
        \item 焦半径：$|AF| = x_1 + \dfrac{p}{2}$，$|BF| = x_2 + \dfrac{p}{2}$
        \item 焦点弦长：$|AB| = x_1 + x_2 + p = \dfrac{2p}{\sin^2\theta}$（$\theta$ 为弦与 $x$ 轴夹角）
        \item 通径（过焦点垂直于对称轴的弦）：长为 $2p$
        \item 焦点弦的调和性质：$\dfrac{1}{|AF|} + \dfrac{1}{|BF|} = \dfrac{2}{p}$
    \end{itemize}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[抛物线的定义应用]
    点 $F$ 为抛物线 $C: y^2 = 8x$ 的焦点，$P$ 为 $C$ 上一点，若 $\triangle POF$ 的面积为 $4\sqrt{2}$（$O$ 为坐标原点），求 $|PF|$。
\end{example}

\textbf{解.} 抛物线 $C: y^2 = 8x$ 中，$2p = 8$，所以 $p = 4$，焦点 $F(2, 0)$，准线方程为 $x = -2$。

设 $P(x_0, y_0)$，则 $y_0^2 = 8x_0$。

$\triangle POF$ 的面积为
\[
    S = \frac{1}{2}|OF| \cdot |y_0| = \frac{1}{2} \times 2 \times |y_0| = |y_0| = 4\sqrt{2}
\]

所以 $y_0 = \pm 4\sqrt{2}$，代入抛物线方程得 $x_0 = \dfrac{y_0^2}{8} = \dfrac{32}{8} = 4$。

由抛物线定义，$|PF| = x_0 + \dfrac{p}{2} = 4 + 2 = 6$。

\begin{realexam}[2026届奉贤二模]
    设 $m > 2$，抛物线 $C: x^2 = 2my$ 上的点 $P$ 到 $C$ 的焦点的距离为 $5$，点 $P$ 到 $y$ 轴的距离为 $3$，求 $m$ 的值。
\end{realexam}

\textbf{解.} 抛物线 $C: x^2 = 2my$（$m > 0$）开口向上，焦点为 $F\left(0, \dfrac{m}{2}\right)$，准线为 $y = -\dfrac{m}{2}$。

设 $P(x_0, y_0)$，由点 $P$ 到 $y$ 轴的距离为 $3$，得 $|x_0| = 3$。

由点 $P$ 在抛物线上，$x_0^2 = 2my_0$，即 $9 = 2my_0$，所以 $y_0 = \dfrac{9}{2m}$。

由抛物线定义，点 $P$ 到焦点的距离等于到准线的距离：
\[
    |PF| = y_0 + \frac{m}{2} = \frac{9}{2m} + \frac{m}{2} = 5
\]

解方程 $\dfrac{9}{2m} + \dfrac{m}{2} = 5$，得 $9 + m^2 = 10m$，即 $m^2 - 10m + 9 = 0$。

解得 $m = 1$ 或 $m = 9$。

由条件 $m > 2$，所以 $m = 9$。

\begin{example}[抛物线的最值问题]
    设 $M(x, y)$ 为抛物线 $y^2 = 4x$ 上任意一点，若 $x + 2y + m$ 的最小值为 $1$，求 $m$ 的值。
\end{example}

\textbf{解.} 由 $M(x, y)$ 在抛物线 $y^2 = 4x$ 上，得 $x = \dfrac{y^2}{4}$，$y \in \mathbb{R}$。

所以
\[
    x + 2y + m = \frac{y^2}{4} + 2y + m = \frac{1}{4}(y^2 + 8y) + m = \frac{1}{4}(y + 4)^2 - 4 + m
\]

当 $y = -4$ 时，取得最小值 $-4 + m = 1$。

所以 $m = 5$。

\begin{realexam}[2025届上海二模]
    已知抛物线 $\Gamma: x^2 = 4y$，过点 $P(a, b)$ 的直线 $l$ 与抛物线 $\Gamma$ 交于点 $A$、$B$，与 $y$ 轴交于点 $C$。
    \begin{enumerate}
        \item 若点 $A$ 位于第一象限，且点 $A$ 到抛物线 $\Gamma$ 的焦点的距离等于 $3$，求点 $A$ 的坐标；
        \item 若抛物线 $\Gamma$ 上存在定点 $D$ 使得满足题意的点 $A$、$B$ 都有 $DA \perp DB$，求 $a$、$b$ 满足的关系式。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 抛物线 $\Gamma: x^2 = 4y$ 的焦点为 $F(0, 1)$，准线为 $y = -1$。

设 $A(x, y)$（$x > 0, y > 0$），由抛物线定义，点 $A$ 到焦点的距离等于到准线的距离：
\[
    y + 1 = 3 \Rightarrow y = 2
\]

代入抛物线方程得 $x^2 = 8$，所以 $x = 2\sqrt{2}$（取正值）。

故点 $A$ 的坐标为 $(2\sqrt{2}, 2)$。

(2) 设 $D\left(x_0, \dfrac{x_0^2}{4}\right)$，直线 $l$ 的斜率存在，设其方程为 $y - b = k(x - a)$。

代入 $x^2 = 4y$ 得 $x^2 - 4kx + 4ka - 4b = 0$。

设 $A\left(x_1, \dfrac{x_1^2}{4}\right)$，$B\left(x_2, \dfrac{x_2^2}{4}\right)$，则 $x_1 + x_2 = 4k$，$x_1x_2 = 4ka - 4b$。

由 $DA \perp DB$，得 $\overrightarrow{DA} \cdot \overrightarrow{DB} = 0$。

计算得
\[
    (x_1 - x_0)(x_2 - x_0) + \left(\frac{x_1^2}{4} - \frac{x_0^2}{4}\right)\left(\frac{x_2^2}{4} - \frac{x_0^2}{4}\right) = 0
\]

化简整理，利用韦达定理，可得 $a$、$b$ 满足的关系式为 $b = \dfrac{a^2}{4} + 4$。

\begin{example}[抛物线与圆的综合]
    已知点 $P$ 是抛物线 $y^2 = 8x$ 上一动点，点 $Q$ 在圆 $(x - 5)^2 + y^2 = 1$ 上运动，求 $|PQ|$ 的最小值。
\end{example}

\textbf{解.} 设抛物线 $y^2 = 8x$ 上的点 $P$ 坐标为 $\left(\dfrac{y^2}{8}, y\right)$。

圆 $(x - 5)^2 + y^2 = 1$ 的圆心为 $C(5, 0)$，半径 $r = 1$。

点 $P$ 到圆心 $C$ 的距离
\[
    |PC| = \sqrt{\left(\frac{y^2}{8} - 5\right)^2 + y^2} = \sqrt{\frac{y^4}{64} - \frac{5y^2}{4} + 25 + y^2} = \sqrt{\frac{y^4}{64} - \frac{y^2}{4} + 25}
\]

令 $t = y^2$（$t \geq 0$），则
\[
    |PC| = \sqrt{\frac{t^2}{64} - \frac{t}{4} + 25} = \sqrt{\frac{1}{64}(t - 8)^2 + 24}
\]

当 $t = 8$ 时，$|PC|$ 取得最小值 $\sqrt{24} = 2\sqrt{6}$。

所以 $|PQ|$ 的最小值为 $|PC|_{\min} - r = 2\sqrt{6} - 1$。

\begin{example}[抛物线的焦点弦问题]
    过抛物线 $y^2 = 4x$ 的焦点 $F$ 作两条互相垂直的直线 $l_1$、$l_2$，分别交抛物线于 $A$、$B$ 和 $C$、$D$ 四点，求 $|AB| + |CD|$ 的最小值。
\end{example}

\textbf{解.} 抛物线 $y^2 = 4x$ 的焦点为 $F(1, 0)$，$p = 2$。

设直线 $l_1$ 的倾斜角为 $\theta$，则直线 $l_2$ 的倾斜角为 $\theta + \dfrac{\pi}{2}$。

由焦点弦长公式：$|AB| = \dfrac{2p}{\sin^2\theta} = \dfrac{4}{\sin^2\theta}$，$|CD| = \dfrac{2p}{\sin^2(\theta + \frac{\pi}{2})} = \dfrac{4}{\cos^2\theta}$。

所以
\[
    |AB| + |CD| = \frac{4}{\sin^2\theta} + \frac{4}{\cos^2\theta} = \frac{4}{\sin^2\theta\cos^2\theta} = \frac{16}{\sin^2 2\theta}
\]

当 $\sin^2 2\theta = 1$，即 $\theta = \dfrac{\pi}{4}$ 时，$|AB| + |CD|$ 取得最小值 $16$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{定义转化}：抛物线问题中，常利用定义将点到焦点的距离转化为点到准线的距离，简化计算。
    
    \item \textbf{标准方程识别}：首先将抛物线方程化为标准形式，确定开口方向、焦点位置和准线方程。
    
    \item \textbf{焦半径公式}：对于 $y^2 = 2px$ 上的点 $P(x_0, y_0)$，焦半径 $|PF| = x_0 + \dfrac{p}{2}$，避免用距离公式计算。
    
    \item \textbf{参数方程}：抛物线 $y^2 = 2px$ 的参数方程为 $\begin{cases} x = 2pt^2 \\ y = 2pt \end{cases}$，有时可简化运算。
    
    \item \textbf{设而不求}：涉及弦中点、弦长问题时，常设直线方程，联立后用韦达定理，避免求具体交点坐标。
    
    \item \textbf{几何性质}：抛物线的光学性质（反射性质）和焦点弦性质在解题中经常用到。
\end{enumerate}
