\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

三角恒等变换是三角函数的核心内容，通过公式将不同形式的三角表达式相互转化，为化简、求值、证明提供工具。

\begin{definition}[两角和与差的三角公式]
    \begin{align*}
        \sin(\alpha \pm \beta) &= \sin\alpha\cos\beta \pm \cos\alpha\sin\beta \\
        \cos(\alpha \pm \beta) &= \cos\alpha\cos\beta \mp \sin\alpha\sin\beta \\
        \tan(\alpha \pm \beta) &= \frac{\tan\alpha \pm \tan\beta}{1 \mp \tan\alpha\tan\beta}
    \end{align*}
\end{definition}

\begin{definition}[二倍角公式]
    \begin{align*}
        \sin 2\alpha &= 2\sin\alpha\cos\alpha \\
        \cos 2\alpha &= \cos^2\alpha - \sin^2\alpha = 2\cos^2\alpha - 1 = 1 - 2\sin^2\alpha \\
        \tan 2\alpha &= \frac{2\tan\alpha}{1 - \tan^2\alpha}
    \end{align*}
\end{definition}

\begin{definition}[半角公式]
    \begin{align*}
        \sin\frac{\alpha}{2} &= \pm\sqrt{\frac{1 - \cos\alpha}{2}} \\
        \cos\frac{\alpha}{2} &= \pm\sqrt{\frac{1 + \cos\alpha}{2}} \\
        \tan\frac{\alpha}{2} &= \pm\sqrt{\frac{1 - \cos\alpha}{1 + \cos\alpha}} = \frac{\sin\alpha}{1 + \cos\alpha} = \frac{1 - \cos\alpha}{\sin\alpha}
    \end{align*}
    符号由 $\dfrac{\alpha}{2}$ 所在象限确定。
\end{definition}

\begin{property}[降幂公式]
    由二倍角公式变形得到：
    \[
        \sin^2\alpha = \frac{1 - \cos 2\alpha}{2}, \quad \cos^2\alpha = \frac{1 + \cos 2\alpha}{2}.
    \]
\end{property}

\begin{property}[辅助角公式]
    $a\sin\alpha + b\cos\alpha = \sqrt{a^2 + b^2}\sin(\alpha + \varphi)$，其中 $\tan\varphi = \dfrac{b}{a}$，$\varphi$ 的终边过点 $(a, b)$。
    
    常见形式：
    \begin{itemize}
        \item $\sin\alpha \pm \cos\alpha = \sqrt{2}\sin\left(\alpha \pm \dfrac{\pi}{4}\right)$
        \item $\sqrt{3}\sin\alpha \pm \cos\alpha = 2\sin\left(\alpha \pm \dfrac{\pi}{6}\right)$
    \end{itemize}
\end{property}

\begin{property}[积化和差与和差化积]
    \begin{align*}
        \sin\alpha\cos\beta &= \frac{1}{2}[\sin(\alpha + \beta) + \sin(\alpha - \beta)] \\
        \cos\alpha\sin\beta &= \frac{1}{2}[\sin(\alpha + \beta) - \sin(\alpha - \beta)] \\
        \cos\alpha\cos\beta &= \frac{1}{2}[\cos(\alpha + \beta) + \cos(\alpha - \beta)] \\
        \sin\alpha\sin\beta &= -\frac{1}{2}[\cos(\alpha + \beta) - \cos(\alpha - \beta)]
    \end{align*}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[两角和差公式的应用]
    已知 $\sin\alpha = \dfrac{3}{5}$，$\alpha \in \left(\dfrac{\pi}{2}, \pi\right)$，$\cos\beta = -\dfrac{5}{13}$，$\beta \in \left(\pi, \dfrac{3\pi}{2}\right)$，求 $\sin(\alpha + \beta)$ 的值。
\end{example}

\textbf{解.} 由 $\sin\alpha = \dfrac{3}{5}$，$\alpha \in \left(\dfrac{\pi}{2}, \pi\right)$，得
\[
    \cos\alpha = -\sqrt{1 - \sin^2\alpha} = -\sqrt{1 - \frac{9}{25}} = -\frac{4}{5}.
\]

由 $\cos\beta = -\dfrac{5}{13}$，$\beta \in \left(\pi, \dfrac{3\pi}{2}\right)$，得
\[
    \sin\beta = -\sqrt{1 - \cos^2\beta} = -\sqrt{1 - \frac{25}{169}} = -\frac{12}{13}.
\]

由两角和公式：
\begin{align*}
    \sin(\alpha + \beta) &= \sin\alpha\cos\beta + \cos\alpha\sin\beta \\
    &= \frac{3}{5} \times \left(-\frac{5}{13}\right) + \left(-\frac{4}{5}\right) \times \left(-\frac{12}{13}\right) \\
    &= -\frac{15}{65} + \frac{48}{65} = \frac{33}{65}.
\end{align*}

\begin{example}[二倍角公式的应用]
    已知 $\sin\alpha + \cos\alpha = \dfrac{1}{3}$，$\alpha \in (0, \pi)$，求 $\sin 2\alpha$ 的值。
\end{example}

\textbf{解.} 将已知等式两边平方：
\[
    (\sin\alpha + \cos\alpha)^2 = \frac{1}{9}.
\]

展开得：
\[
    \sin^2\alpha + 2\sin\alpha\cos\alpha + \cos^2\alpha = \frac{1}{9}.
\]

即 $1 + \sin 2\alpha = \dfrac{1}{9}$，所以 $\sin 2\alpha = -\dfrac{8}{9}$。

\begin{example}[辅助角公式的应用]
    求函数 $f(x) = \sin x + \sqrt{3}\cos x$ 的最大值和最小值。
\end{example}

\textbf{解.} 利用辅助角公式：
\[
    f(x) = \sin x + \sqrt{3}\cos x = 2\left(\frac{1}{2}\sin x + \frac{\sqrt{3}}{2}\cos x\right) = 2\sin\left(x + \frac{\pi}{3}\right).
\]

因为 $\sin\left(x + \dfrac{\pi}{3}\right) \in [-1, 1]$，所以
\[
    f(x)_{\max} = 2, \quad f(x)_{\min} = -2.
\]

\begin{realexam}[2026 奉贤高三二模]
    在平面直角坐标系 $xOy$ 中，点 $A(\cos\theta, \sin\theta)$，$B(-\sin\theta, \cos\theta)$，$\theta \in [0, 2\pi)$。若点 $P(x, y)$ 满足：$\overrightarrow{OP} \cdot \overrightarrow{OA} = 1$，$\overrightarrow{OP} \cdot \overrightarrow{OB} = 2$，则 $y$ 的最大值是 \underline{\hspace{2cm}}。
\end{realexam}

\textbf{解.} 由题意得
\[
    \begin{cases}
        x\cos\theta + y\sin\theta = 1 \\
        -x\sin\theta + y\cos\theta = 2
    \end{cases}
\]

解得 $y = \sin\theta + 2\cos\theta = \sqrt{5}\sin(\theta + \varphi)$，其中 $\tan\varphi = 2$。

当 $\sin(\theta + \varphi) = 1$ 时，$y$ 取最大值 $\sqrt{5}$。

\begin{realexam}[练习]
    化简：$\dfrac{\sin 7^{\circ} + \cos 15^{\circ}\sin 8^{\circ}}{\cos 7^{\circ} - \sin 15^{\circ}\sin 8^{\circ}}$。
\end{realexam}

\textbf{解.} 注意到 $7^{\circ} = 15^{\circ} - 8^{\circ}$，利用两角差公式：
\begin{align*}
    \text{分子} &= \sin(15^{\circ} - 8^{\circ}) + \cos 15^{\circ}\sin 8^{\circ} \\
    &= \sin 15^{\circ}\cos 8^{\circ} - \cos 15^{\circ}\sin 8^{\circ} + \cos 15^{\circ}\sin 8^{\circ} \\
    &= \sin 15^{\circ}\cos 8^{\circ}.
\end{align*}

\begin{align*}
    \text{分母} &= \cos(15^{\circ} - 8^{\circ}) - \sin 15^{\circ}\sin 8^{\circ} \\
    &= \cos 15^{\circ}\cos 8^{\circ} + \sin 15^{\circ}\sin 8^{\circ} - \sin 15^{\circ}\sin 8^{\circ} \\
    &= \cos 15^{\circ}\cos 8^{\circ}.
\end{align*}

原式 $= \dfrac{\sin 15^{\circ}\cos 8^{\circ}}{\cos 15^{\circ}\cos 8^{\circ}} = \tan 15^{\circ} = 2 - \sqrt{3}$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{角的拆分与拼凑}：将未知角表示为已知角的和差倍分，如 $2\alpha = (\alpha + \beta) + (\alpha - \beta)$。
    \item \textbf{"1"的代换}：利用 $\sin^2\alpha + \cos^2\alpha = 1$ 进行齐次化处理。
    \item \textbf{弦切互化}：齐次式可同除 $\cos\alpha$ 转化为关于 $\tan\alpha$ 的表达式。
    \item \textbf{辅助角公式}：$a\sin\alpha + b\cos\alpha$ 型函数求最值时，必用辅助角公式化为单角三角函数。
    \item \textbf{降幂升角}：遇到 $\sin^2\alpha$、$\cos^2\alpha$ 时，考虑用降幂公式转化为二倍角。
    \item \textbf{和积互化}：和差化积与积化和差在特定结构下可简化计算。
\end{enumerate}
