\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

三角函数的最值问题是高考的重点和难点，涉及代数变形、三角恒等变换、函数性质等多个知识点的综合运用。

\begin{definition}[三角函数最值的基本类型]
    三角函数最值问题常见类型：
    \begin{enumerate}
        \item \textbf{基本型}：$y = A\sin(\omega x + \varphi) + B$ 或 $y = A\cos(\omega x + \varphi) + B$，利用正弦、余弦函数的有界性求解；
        \item \textbf{辅助角型}：$y = a\sin x + b\cos x$，利用辅助角公式化为 $y = \sqrt{a^2 + b^2}\sin(x + \varphi)$；
        \item \textbf{二次型}：$y = a\sin^2 x + b\sin x + c$ 或 $y = a\cos^2 x + b\cos x + c$，利用换元法转化为二次函数；
        \item \textbf{分式型}：$y = \dfrac{a\sin x + b}{c\sin x + d}$ 或含有三角函数的分式，利用分离常数或有界性求解；
        \item \textbf{复合型}：涉及多个三角函数或与其他函数（如指数、对数）复合的综合问题。
    \end{enumerate}
\end{definition}

\begin{property}[辅助角公式]
    $a\sin\alpha + b\cos\alpha = \sqrt{a^2 + b^2}\sin(\alpha + \varphi)$，其中 $\tan\varphi = \dfrac{b}{a}$，$\varphi$ 的终边过点 $(a, b)$。
    
    常见形式：
    \begin{itemize}
        \item $\sin\alpha \pm \cos\alpha = \sqrt{2}\sin\left(\alpha \pm \dfrac{\pi}{4}\right)$
        \item $\sqrt{3}\sin\alpha \pm \cos\alpha = 2\sin\left(\alpha \pm \dfrac{\pi}{6}\right)$
        \item $\sin\alpha \pm \sqrt{3}\cos\alpha = 2\sin\left(\alpha \pm \dfrac{\pi}{3}\right)$
    \end{itemize}
\end{property}

\begin{property}[换元法求最值]
    对于 $y = a\sin^2 x + b\sin x + c$（或关于 $\cos x$ 的类似形式）：
    \begin{enumerate}
        \item 令 $t = \sin x$（或 $t = \cos x$），则 $t \in [-1, 1]$；
        \item 转化为二次函数 $y = at^2 + bt + c$ 在区间 $[-1, 1]$ 上的最值问题；
        \item 注意对称轴 $t = -\dfrac{b}{2a}$ 与区间 $[-1, 1]$ 的位置关系。
    \end{enumerate}
\end{property}

\begin{property}[基本不等式在三角函数中的应用]
    当三角函数表达式可以化为积为定值或和为定值的形式时，可利用基本不等式求最值：
    \begin{itemize}
        \item 若 $a, b > 0$，则 $a + b \geqslant 2\sqrt{ab}$（当且仅当 $a = b$ 时取等号）；
        \item 注意验证等号成立的条件是否在定义域内。
    \end{itemize}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[辅助角公式求最值]
    求函数 $f(x) = \sin x + \sqrt{3}\cos x$ 在 $x \in \left[0, \dfrac{\pi}{2}\right]$ 上的最大值和最小值。
\end{example}

\textbf{解.} 利用辅助角公式：
\[
    f(x) = \sin x + \sqrt{3}\cos x = 2\left(\frac{1}{2}\sin x + \frac{\sqrt{3}}{2}\cos x\right) = 2\sin\left(x + \frac{\pi}{3}\right).
\]

当 $x \in \left[0, \dfrac{\pi}{2}\right]$ 时，$x + \dfrac{\pi}{3} \in \left[\dfrac{\pi}{3}, \dfrac{5\pi}{6}\right]$。

当 $x + \dfrac{\pi}{3} = \dfrac{\pi}{2}$，即 $x = \dfrac{\pi}{6}$ 时，$f(x)_{\max} = 2$。

当 $x + \dfrac{\pi}{3} = \dfrac{5\pi}{6}$，即 $x = \dfrac{\pi}{2}$ 时，$f(x)_{\min} = 2 \times \dfrac{1}{2} = 1$。

\begin{example}[换元法求最值]
    求函数 $y = \cos^2 x + 2\sin x - 2$ 的最大值和最小值。
\end{example}

\textbf{解.} 利用 $\cos^2 x = 1 - \sin^2 x$，得：
\[
    y = 1 - \sin^2 x + 2\sin x - 2 = -\sin^2 x + 2\sin x - 1.
\]

令 $t = \sin x$，则 $t \in [-1, 1]$，$y = -t^2 + 2t - 1 = -(t - 1)^2$。

当 $t = 1$ 时，$y_{\max} = 0$。

当 $t = -1$ 时，$y_{\min} = -(-1 - 1)^2 = -4$。

\begin{example}[分式型三角函数最值]
    求函数 $y = \dfrac{2\sin x + 1}{\sin x - 1}$ 的值域。
\end{example}

\textbf{解.} 令 $t = \sin x$，则 $t \in [-1, 1]$，且 $t \neq 1$（分母不为零）。

$y = \dfrac{2t + 1}{t - 1} = \dfrac{2(t - 1) + 3}{t - 1} = 2 + \dfrac{3}{t - 1}$。

当 $t \in [-1, 1)$ 时，$t - 1 \in [-2, 0)$，$\dfrac{3}{t - 1} \in \left(-\infty, -\dfrac{3}{2}\right]$。

所以 $y \in \left(-\infty, \dfrac{1}{2}\right]$。

\begin{example}[综合应用——三角函数与向量]
    在平面直角坐标系 $xOy$ 中，点 $A(\cos\theta, \sin\theta)$，$B(-\sin\theta, \cos\theta)$，$\theta \in [0, 2\pi)$。若点 $P(x, y)$ 满足：$\overrightarrow{OP} \cdot \overrightarrow{OA} = 1$，$\overrightarrow{OP} \cdot \overrightarrow{OB} = 2$，求 $y$ 的最大值。
\end{example}

\textbf{解.} 由题意得：
\[
    \begin{cases}
        x\cos\theta + y\sin\theta = 1 \\
        -x\sin\theta + y\cos\theta = 2
    \end{cases}
\]

将两式分别乘以 $\cos\theta$ 和 $\sin\theta$ 后相加，解得：
\[
    y = \sin\theta + 2\cos\theta = \sqrt{5}\sin(\theta + \varphi),
\]
其中 $\tan\varphi = 2$。

当 $\sin(\theta + \varphi) = 1$ 时，$y$ 取最大值 $\sqrt{5}$。

\begin{realexam}[2026 松江高三一模]
    已知函数 $y = A\sin(\omega x + \varphi)$（$A, \omega > 0$，$0 < \varphi < \pi$）的周期为 $\pi$，在 $x = \dfrac{\pi}{6}$ 时取到最大值 4，记 $y = f(x)$。
    \begin{enumerate}
        \item 求 $f(x)$ 的解析式；
        \item 求函数 $y = 2\sin^2 x + f(x)$，$x \in \left[-\dfrac{\pi}{6}, \dfrac{\pi}{3}\right]$ 的最大值，并求出取得最大值时 $x$ 的值。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 由周期 $T = \pi$，得 $\omega = \dfrac{2\pi}{T} = 2$。

由最大值为 4，得 $A = 4$。

当 $x = \dfrac{\pi}{6}$ 时取最大值，即 $2 \times \dfrac{\pi}{6} + \varphi = \dfrac{\pi}{2} + 2k\pi$。

所以 $\varphi = \dfrac{\pi}{6}$（满足 $0 < \varphi < \pi$）。

因此，$f(x) = 4\sin\left(2x + \dfrac{\pi}{6}\right)$。

(2) $y = 2\sin^2 x + 4\sin\left(2x + \dfrac{\pi}{6}\right)$。

利用 $2\sin^2 x = 1 - \cos 2x$，得：
\[
    y = 1 - \cos 2x + 4\left(\sin 2x\cos\frac{\pi}{6} + \cos 2x\sin\frac{\pi}{6}\right) = 1 - \cos 2x + 2\sqrt{3}\sin 2x + 2\cos 2x = 1 + 2\sqrt{3}\sin 2x + \cos 2x.
\]

利用辅助角公式：$2\sqrt{3}\sin 2x + \cos 2x = \sqrt{13}\sin(2x + \alpha)$，其中 $\tan\alpha = \dfrac{1}{2\sqrt{3}} = \dfrac{\sqrt{3}}{6}$。

当 $x \in \left[-\dfrac{\pi}{6}, \dfrac{\pi}{3}\right]$ 时，$2x \in \left[-\dfrac{\pi}{3}, \dfrac{2\pi}{3}\right]$。

当 $2x = \dfrac{\pi}{2}$，即 $x = \dfrac{\pi}{4}$ 时，$\sin 2x = 1$，$\cos 2x = 0$，$y = 1 + 2\sqrt{3}$。

当 $2x = -\dfrac{\pi}{3}$，即 $x = -\dfrac{\pi}{6}$ 时，$\sin 2x = -\dfrac{\sqrt{3}}{2}$，$\cos 2x = \dfrac{1}{2}$，$y = 1 - 3 + \dfrac{1}{2} = -\dfrac{3}{2}$。

当 $2x = \dfrac{2\pi}{3}$，即 $x = \dfrac{\pi}{3}$ 时，$\sin 2x = \dfrac{\sqrt{3}}{2}$，$\cos 2x = -\dfrac{1}{2}$，$y = 1 + 3 - \dfrac{1}{2} = \dfrac{7}{2}$。

比较得，当 $x = \dfrac{\pi}{3}$ 时，$y_{\max} = \dfrac{7}{2}$。

\begin{realexam}[2025 上海秋考]
    已知函数 $f(x) = 2\sin(\omega x)$，其中常数 $\omega > 0$。
    \begin{enumerate}
        \item 若 $y = f(x)$ 在 $\left[-\dfrac{\pi}{4}, \dfrac{2\pi}{3}\right]$ 上单调递增，求 $\omega$ 的取值范围；
        \item 令 $\omega = 2$，将函数 $y = f(x)$ 的图像向左平移 $\dfrac{\pi}{6}$ 个单位，再向上平移 1 个单位，得到函数 $y = g(x)$ 的图像。求 $g(x)$ 在区间 $[a, b]$（$a, b \in \mathbb{R}$ 且 $a < b$）上至少含有 30 个零点时，$b - a$ 的最小值。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) $f(x) = 2\sin(\omega x)$ 的单调递增区间为 $\left[\dfrac{2k\pi}{\omega} - \dfrac{\pi}{2\omega}, \dfrac{2k\pi}{\omega} + \dfrac{\pi}{2\omega}\right]$（$k \in \mathbb{Z}$）。

取 $k = 0$，单调递增区间为 $\left[-\dfrac{\pi}{2\omega}, \dfrac{\pi}{2\omega}\right]$。

要使 $f(x)$ 在 $\left[-\dfrac{\pi}{4}, \dfrac{2\pi}{3}\right]$ 上单调递增，需满足：
\[
    \begin{cases}
        -\dfrac{\pi}{2\omega} \leqslant -\dfrac{\pi}{4} \\
        \dfrac{\pi}{2\omega} \geqslant \dfrac{2\pi}{3}
    \end{cases}
    \Rightarrow
    \begin{cases}
        \omega \leqslant 2 \\
        \omega \leqslant \dfrac{3}{4}
    \end{cases}
\]

结合 $\omega > 0$，得 $\omega \in \left(0, \dfrac{3}{4}\right]$。

(2) 当 $\omega = 2$ 时，$f(x) = 2\sin(2x)$。

向左平移 $\dfrac{\pi}{6}$ 个单位，得 $y = 2\sin\left(2\left(x + \dfrac{\pi}{6}\right)\right) = 2\sin\left(2x + \dfrac{\pi}{3}\right)$。

再向上平移 1 个单位，得 $g(x) = 2\sin\left(2x + \dfrac{\pi}{3}\right) + 1$。

令 $g(x) = 0$，即 $\sin\left(2x + \dfrac{\pi}{3}\right) = -\dfrac{1}{2}$。

解得 $2x + \dfrac{\pi}{3} = 2k\pi - \dfrac{\pi}{6}$ 或 $2x + \dfrac{\pi}{3} = 2k\pi + \dfrac{7\pi}{6}$（$k \in \mathbb{Z}$）。

即 $x = k\pi - \dfrac{\pi}{4}$ 或 $x = k\pi + \dfrac{5\pi}{12}$。

相邻零点之间的距离为 $\dfrac{5\pi}{12} - \left(-\dfrac{\pi}{4}\right) = \dfrac{2\pi}{3}$ 或 $\pi - \dfrac{2\pi}{3} = \dfrac{\pi}{3}$。

要使 $[a, b]$ 上至少含有 30 个零点且 $b - a$ 最小，取间隔最紧凑的排列：
从 $x = -\dfrac{\pi}{4}$ 开始，经过 29 个间隔（15 个长间隔 $\dfrac{2\pi}{3}$ 和 14 个短间隔 $\dfrac{\pi}{3}$），到达第 30 个零点。

$b - a$ 的最小值为 $14 \times \dfrac{\pi}{3} + 15 \times \dfrac{2\pi}{3} = \dfrac{14\pi + 30\pi}{3} = \dfrac{44\pi}{3}$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{辅助角公式}：$a\sin x + b\cos x = \sqrt{a^2 + b^2}\sin(x + \varphi)$ 是求三角函数最值的核心工具，务必熟练掌握。
    \item \textbf{换元思想}：对于含有 $\sin^2 x$、$\cos^2 x$、$\sin x\cos x$ 的表达式，可利用降幂公式或换元转化为代数函数。
    \item \textbf{有界性应用}：$|\sin x| \leqslant 1$，$|\cos x| \leqslant 1$ 是求三角函数最值的重要依据。
    \item \textbf{定义域优先}：求最值时必须先确定自变量的取值范围，特别是由实际问题或复合函数确定的定义域。
    \item \textbf{分类讨论}：对于含参数的三角函数最值问题，注意对参数的不同取值范围进行分类讨论。
    \item \textbf{综合应用}：三角函数与向量、解三角形、函数与方程等知识的综合问题，需要灵活运用各种数学工具。
    \item \textbf{数形结合}：画出示意图，利用三角函数的周期性、对称性简化计算。
\end{enumerate}
