\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

直线的方程与位置关系是解析几何的基础内容，掌握直线的各种方程形式及位置关系的判定方法是解决解析几何问题的关键。

\begin{definition}[直线的倾斜角与斜率]
    在平面直角坐标系中，对于一条与 $x$ 轴相交的直线，把 $x$ 轴正方向按逆时针方向绕着交点旋转到和直线重合所成的角，叫做直线的\textbf{倾斜角}，记作 $\alpha$，其中 $\alpha \in [0, \pi)$。
    
    当 $\alpha \neq \dfrac{\pi}{2}$ 时，定义直线的\textbf{斜率}为
    \[
        k = \tan\alpha.
    \]
    过两点 $P_1(x_1, y_1)$、$P_2(x_2, y_2)$（$x_1 \neq x_2$）的直线的斜率为
    \[
        k = \frac{y_2 - y_1}{x_2 - x_1}.
    \]
\end{definition}

\begin{property}[直线的方程形式]
    直线方程有以下几种常用形式：
    \begin{enumerate}
        \item \textbf{点斜式}：$y - y_0 = k(x - x_0)$，适用于已知斜率和一点；
        \item \textbf{斜截式}：$y = kx + b$，适用于已知斜率和 $y$ 轴截距；
        \item \textbf{两点式}：$\dfrac{y - y_1}{y_2 - y_1} = \dfrac{x - x_1}{x_2 - x_1}$，适用于已知两点；
        \item \textbf{截距式}：$\dfrac{x}{a} + \dfrac{y}{b} = 1$（$ab \neq 0$），适用于已知两轴截距；
        \item \textbf{一般式}：$Ax + By + C = 0$（$A^2 + B^2 \neq 0$），适用于所有情况。
    \end{enumerate}
\end{property}

\begin{property}[两直线的位置关系]
    设直线 $l_1: A_1x + B_1y + C_1 = 0$，$l_2: A_2x + B_2y + C_2 = 0$：
    \begin{itemize}
        \item \textbf{平行}：$l_1 \parallel l_2 \Leftrightarrow A_1B_2 = A_2B_1$ 且 $A_1C_2 \neq A_2C_1$（或 $B_1C_2 \neq B_2C_1$）；
        \item \textbf{垂直}：$l_1 \perp l_2 \Leftrightarrow A_1A_2 + B_1B_2 = 0$；
        \item \textbf{相交}：$A_1B_2 \neq A_2B_1$；
        \item \textbf{重合}：$\dfrac{A_1}{A_2} = \dfrac{B_1}{B_2} = \dfrac{C_1}{C_2}$。
    \end{itemize}
\end{property}

\begin{property}[点到直线的距离]
    点 $P(x_0, y_0)$ 到直线 $Ax + By + C = 0$ 的距离为
    \[
        d = \frac{|Ax_0 + By_0 + C|}{\sqrt{A^2 + B^2}}.
    \]
    
    两条平行直线 $Ax + By + C_1 = 0$ 与 $Ax + By + C_2 = 0$ 之间的距离为
    \[
        d = \frac{|C_1 - C_2|}{\sqrt{A^2 + B^2}}.
    \]
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[求直线方程]
    已知直线 $l$ 经过点 $P(2, -3)$，且分别满足下列条件，求直线 $l$ 的方程：
    \begin{enumerate}
        \item 倾斜角为 $135°$；
        \item 与直线 $2x - y + 3 = 0$ 平行；
        \item 与直线 $x + 2y - 1 = 0$ 垂直。
    \end{enumerate}
\end{example}

\textbf{解.} (1) 倾斜角 $\alpha = 135°$，则斜率 $k = \tan 135° = -1$。
由点斜式得直线方程为 $y + 3 = -(x - 2)$，即 $\boxed{x + y + 1 = 0}$。

(2) 与直线 $2x - y + 3 = 0$ 平行，则斜率相同，$k = 2$。
由点斜式得 $y + 3 = 2(x - 2)$，即 $\boxed{2x - y - 7 = 0}$。

(3) 直线 $x + 2y - 1 = 0$ 的斜率为 $-\dfrac{1}{2}$，与之垂直的直线斜率为 $k = 2$。
由点斜式得 $y + 3 = 2(x - 2)$，即 $\boxed{2x - y - 7 = 0}$。

\begin{realexam}[2026 奉贤二模]
    若直线 $(a - 1)x + y - 1 = 0$ 与直线 $3x - y = 0$ 平行，则实数 $a$ 的值为\_\_\_\_\_。
\end{realexam}

\textbf{解.} 已知直线 $(a - 1)x + y - 1 = 0$ 与直线 $3x - y = 0$ 平行，

$\therefore$ 两直线斜率相等，即 $1 - a = 3$，解得 $a = -2$。

$\because$ 直线 $(a - 1)x + y - 1 = 0$ 的截距为 $1$，直线 $3x - y = 0$ 的截距为 $0$，不相等，$\therefore a = -2$。

故答案为 $\boxed{-2}$。

\begin{example}[直线位置关系的判定]
    已知直线 $l_1: (m + 2)x + (m + 3)y - 5 = 0$ 和 $l_2: 6x + (2m - 1)y - 5 = 0$，求实数 $m$ 的值，使得：
    \begin{enumerate}
        \item $l_1$ 与 $l_2$ 相交；
        \item $l_1 \parallel l_2$；
        \item $l_1 \perp l_2$。
    \end{enumerate}
\end{example}

\textbf{解.} (1) $l_1$ 与 $l_2$ 相交 $\Leftrightarrow (m + 2)(2m - 1) \neq 6(m + 3)$

即 $2m^2 + 3m - 2 \neq 6m + 18$，整理得 $2m^2 - 3m - 20 \neq 0$

解得 $(2m + 5)(m - 4) \neq 0$，即 $m \neq -\dfrac{5}{2}$ 且 $m \neq 4$。

(2) $l_1 \parallel l_2 \Leftrightarrow (m + 2)(2m - 1) = 6(m + 3)$ 且两直线不重合

由 $(m + 2)(2m - 1) = 6(m + 3)$ 得 $m = -\dfrac{5}{2}$ 或 $m = 4$。

当 $m = -\dfrac{5}{2}$ 时，$l_1: -\dfrac{1}{2}x + \dfrac{1}{2}y - 5 = 0$，$l_2: 6x - 6y - 5 = 0$，不重合，满足平行；

当 $m = 4$ 时，$l_1: 6x + 7y - 5 = 0$，$l_2: 6x + 7y - 5 = 0$，重合，舍去。

故 $m = \boxed{-\dfrac{5}{2}}$。

(3) $l_1 \perp l_2 \Leftrightarrow 6(m + 2) + (2m - 1)(m + 3) = 0$

即 $6m + 12 + 2m^2 + 5m - 3 = 0$，整理得 $2m^2 + 11m + 9 = 0$

解得 $(2m + 9)(m + 1) = 0$，即 $m = -\dfrac{9}{2}$ 或 $m = -1$。

故 $m = \boxed{-\dfrac{9}{2}}$ 或 $\boxed{-1}$。

\begin{realexam}[2025 上海高考真题汇编]
    已知点 $P(1, 2)$，求过点 $P$ 且与原点距离最大的直线方程，并求最大距离。
\end{realexam}

\textbf{解.} 设过点 $P(1, 2)$ 的直线为 $l$。

当直线 $l$ 与 $OP$ 垂直时，原点到直线 $l$ 的距离最大（其中 $O$ 为原点）。

$\because$ $k_{OP} = \dfrac{2 - 0}{1 - 0} = 2$，$\therefore$ 直线 $l$ 的斜率 $k = -\dfrac{1}{2}$。

直线 $l$ 的方程为 $y - 2 = -\dfrac{1}{2}(x - 1)$，即 $\boxed{x + 2y - 5 = 0}$。

最大距离为 $|OP| = \sqrt{1^2 + 2^2} = \boxed{\sqrt{5}}$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{选择适当的方程形式}：已知斜率时用点斜式或斜截式；已知两点时用两点式；已知截距时用截距式；一般情况用一般式。
    \item \textbf{位置关系的判定}：
    \begin{itemize}
        \item 平行：斜率相等且截距不等（$A_1B_2 = A_2B_1$ 且 $A_1C_2 \neq A_2C_1$）；
        \item 垂直：斜率乘积为 $-1$（$A_1A_2 + B_1B_2 = 0$）；
        \item 注意讨论斜率不存在的情况。
    \end{itemize}
    \item \textbf{距离公式应用}：点到直线距离公式中，注意将直线化为一般式；两平行线距离要求 $x$、$y$ 系数相同。
    \item \textbf{对称问题}：点关于直线对称、直线关于直线对称等问题，关键是利用垂直和平分的条件建立方程。
    \item \textbf{最值问题}：涉及距离最值的问题，常利用几何意义（如垂线段最短）或代数方法（如判别式法）求解。
\end{enumerate}
