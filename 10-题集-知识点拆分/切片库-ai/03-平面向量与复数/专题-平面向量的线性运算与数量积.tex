\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

平面向量是沟通代数与几何的重要工具，既有代数运算特征，又有几何直观意义。掌握向量的线性运算和数量积是解决向量问题的核心基础。

\begin{definition}[向量的概念]
    既有大小又有方向的量叫做\textbf{向量}（或矢量）。向量可以用有向线段表示，记作 $\overrightarrow{AB}$ 或 $\boldsymbol{a}$。向量的大小称为向量的\textbf{模}，记作 $|\overrightarrow{AB}|$ 或 $|\boldsymbol{a}|$。
\end{definition}

\begin{definition}[向量的线性运算]
    设 $\boldsymbol{a}$、$\boldsymbol{b}$ 为平面向量，$\lambda$ 为实数：
    \begin{enumerate}
        \item \textbf{加法}：三角形法则或平行四边形法则，满足交换律 $\boldsymbol{a} + \boldsymbol{b} = \boldsymbol{b} + \boldsymbol{a}$ 和结合律 $(\boldsymbol{a} + \boldsymbol{b}) + \boldsymbol{c} = \boldsymbol{a} + (\boldsymbol{b} + \boldsymbol{c})$；
        \item \textbf{减法}：$\boldsymbol{a} - \boldsymbol{b} = \boldsymbol{a} + (-\boldsymbol{b})$；
        \item \textbf{数乘}：$\lambda\boldsymbol{a}$ 是与 $\boldsymbol{a}$ 共线的向量，模为 $|\lambda||\boldsymbol{a}|$，方向由 $\lambda$ 的符号决定。
    \end{enumerate}
\end{definition}

\begin{definition}[平面向量基本定理]
    如果 $\boldsymbol{e}_1$、$\boldsymbol{e}_2$ 是同一平面内的两个不共线向量，那么对于这一平面内的任一向量 $\boldsymbol{a}$，有且只有一对实数 $\lambda_1$、$\lambda_2$，使得
    \[
        \boldsymbol{a} = \lambda_1\boldsymbol{e}_1 + \lambda_2\boldsymbol{e}_2.
    \]
    我们把不共线的向量 $\boldsymbol{e}_1$、$\boldsymbol{e}_2$ 叫做表示这一平面内所有向量的一组\textbf{基底}。
\end{definition}

\begin{definition}[向量的数量积]
    已知两个非零向量 $\boldsymbol{a}$ 与 $\boldsymbol{b}$，它们的夹角为 $\theta$，则数量
    \[
        \boldsymbol{a} \cdot \boldsymbol{b} = |\boldsymbol{a}||\boldsymbol{b}|\cos\theta
    \]
    叫做 $\boldsymbol{a}$ 与 $\boldsymbol{b}$ 的\textbf{数量积}（或内积、点积）。规定：零向量与任何向量的数量积为 $0$。
\end{definition}

\begin{property}[数量积的运算律]
    设 $\boldsymbol{a}$、$\boldsymbol{b}$、$\boldsymbol{c}$ 为向量，$\lambda$ 为实数：
    \begin{itemize}
        \item 交换律：$\boldsymbol{a} \cdot \boldsymbol{b} = \boldsymbol{b} \cdot \boldsymbol{a}$；
        \item 数乘结合律：$(\lambda\boldsymbol{a}) \cdot \boldsymbol{b} = \lambda(\boldsymbol{a} \cdot \boldsymbol{b}) = \boldsymbol{a} \cdot (\lambda\boldsymbol{b})$；
        \item 分配律：$(\boldsymbol{a} + \boldsymbol{b}) \cdot \boldsymbol{c} = \boldsymbol{a} \cdot \boldsymbol{c} + \boldsymbol{b} \cdot \boldsymbol{c}$。
    \end{itemize}
\end{property}

\begin{property}[数量积的重要结论]
    设 $\boldsymbol{a}$、$\boldsymbol{b}$ 为非零向量：
    \begin{itemize}
        \item $|\boldsymbol{a}| = \sqrt{\boldsymbol{a} \cdot \boldsymbol{a}} = \sqrt{\boldsymbol{a}^2}$；
        \item $\cos\theta = \dfrac{\boldsymbol{a} \cdot \boldsymbol{b}}{|\boldsymbol{a}||\boldsymbol{b}|}$（夹角公式）；
        \item $\boldsymbol{a} \perp \boldsymbol{b} \Leftrightarrow \boldsymbol{a} \cdot \boldsymbol{b} = 0$；
        \item $|\boldsymbol{a} \cdot \boldsymbol{b}| \leq |\boldsymbol{a}||\boldsymbol{b}|$（柯西不等式）。
    \end{itemize}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[向量的线性运算]
    在 $\triangle ABC$ 中，点 $D$ 是 $BC$ 的中点，点 $E$ 在 $AD$ 上且 $AE = 2ED$，则 $\overrightarrow{EB} =$（ ）
    
    \textbf{A.} $\dfrac{2}{3}\overrightarrow{AB} - \dfrac{1}{3}\overrightarrow{AC}$ \qquad
    \textbf{B.} $\dfrac{1}{3}\overrightarrow{AB} - \dfrac{2}{3}\overrightarrow{AC}$ \qquad
    \textbf{C.} $\dfrac{5}{6}\overrightarrow{AB} - \dfrac{1}{6}\overrightarrow{AC}$ \qquad
    \textbf{D.} $\dfrac{1}{6}\overrightarrow{AB} - \dfrac{5}{6}\overrightarrow{AC}$
\end{example}

\textbf{解.} 由题意，$\overrightarrow{AD} = \dfrac{1}{2}(\overrightarrow{AB} + \overrightarrow{AC})$。

因为 $AE = 2ED$，所以 $\overrightarrow{AE} = \dfrac{2}{3}\overrightarrow{AD} = \dfrac{1}{3}(\overrightarrow{AB} + \overrightarrow{AC})$。

因此
\[
    \overrightarrow{EB} = \overrightarrow{AB} - \overrightarrow{AE} = \overrightarrow{AB} - \dfrac{1}{3}(\overrightarrow{AB} + \overrightarrow{AC}) = \dfrac{2}{3}\overrightarrow{AB} - \dfrac{1}{3}\overrightarrow{AC}.
\]

故选 \textbf{A}。

\begin{example}[数量积的计算]
    已知 $|\boldsymbol{a}| = 3$，$|\boldsymbol{b}| = 4$，$\boldsymbol{a}$ 与 $\boldsymbol{b}$ 的夹角为 $60^{\circ}$，求：
    \begin{enumerate}
        \item $\boldsymbol{a} \cdot \boldsymbol{b}$；
        \item $(2\boldsymbol{a} + \boldsymbol{b}) \cdot (\boldsymbol{a} - 3\boldsymbol{b})$；
        \item $|2\boldsymbol{a} - \boldsymbol{b}|$。
    \end{enumerate}
\end{example}

\textbf{解.} (1) $\boldsymbol{a} \cdot \boldsymbol{b} = |\boldsymbol{a}||\boldsymbol{b}|\cos 60^{\circ} = 3 \times 4 \times \dfrac{1}{2} = 6$。

(2) 展开计算：
\begin{align*}
    (2\boldsymbol{a} + \boldsymbol{b}) \cdot (\boldsymbol{a} - 3\boldsymbol{b}) &= 2\boldsymbol{a} \cdot \boldsymbol{a} - 6\boldsymbol{a} \cdot \boldsymbol{b} + \boldsymbol{b} \cdot \boldsymbol{a} - 3\boldsymbol{b} \cdot \boldsymbol{b} \\
    &= 2|\boldsymbol{a}|^2 - 5\boldsymbol{a} \cdot \boldsymbol{b} - 3|\boldsymbol{b}|^2 \\
    &= 2 \times 9 - 5 \times 6 - 3 \times 16 \\
    &= 18 - 30 - 48 = -60.
\end{align*}

(3) 先求模的平方：
\[
    |2\boldsymbol{a} - \boldsymbol{b}|^2 = (2\boldsymbol{a} - \boldsymbol{b}) \cdot (2\boldsymbol{a} - \boldsymbol{b}) = 4|\boldsymbol{a}|^2 - 4\boldsymbol{a} \cdot \boldsymbol{b} + |\boldsymbol{b}|^2 = 4 \times 9 - 4 \times 6 + 16 = 36 - 24 + 16 = 28.
\]

所以 $|2\boldsymbol{a} - \boldsymbol{b}| = \sqrt{28} = 2\sqrt{7}$。

\begin{realexam}[2024 上海金山二模]
    已知平面向量 $\overrightarrow{a}$、$\overrightarrow{b}$、$\overrightarrow{c}$ 满足：$|\overrightarrow{a}| = |\overrightarrow{b}| = 1$，$\overrightarrow{a} \cdot \overrightarrow{c} = \overrightarrow{b} \cdot \overrightarrow{c} = 1$，则 $\overrightarrow{a} \cdot \overrightarrow{b} + \overrightarrow{c}^2$ 的最小值为 $\_\_\_\_\_$。
\end{realexam}

\textbf{解.} 设 $\overrightarrow{a}$ 与 $\overrightarrow{b}$ 的夹角为 $\theta$，则 $\overrightarrow{a} \cdot \overrightarrow{b} = \cos\theta$。

由 $\overrightarrow{a} \cdot \overrightarrow{c} = \overrightarrow{b} \cdot \overrightarrow{c} = 1$，得 $\overrightarrow{c} \cdot (\overrightarrow{a} - \overrightarrow{b}) = 0$，即 $\overrightarrow{c} \perp (\overrightarrow{a} - \overrightarrow{b})$。

设 $\overrightarrow{a} = (1, 0)$，$\overrightarrow{b} = (\cos\theta, \sin\theta)$，则 $\overrightarrow{a} - \overrightarrow{b} = (1-\cos\theta, -\sin\theta)$。

由于 $\overrightarrow{c} \perp (\overrightarrow{a} - \overrightarrow{b})$，设 $\overrightarrow{c} = (x, y)$，则 $x(1-\cos\theta) - y\sin\theta = 0$。

又 $\overrightarrow{a} \cdot \overrightarrow{c} = x = 1$，代入得 $1-\cos\theta - y\sin\theta = 0$，所以 $y = \dfrac{1-\cos\theta}{\sin\theta}$。

因此
\[
    \overrightarrow{c}^2 = x^2 + y^2 = 1 + \dfrac{(1-\cos\theta)^2}{\sin^2\theta} = 1 + \dfrac{(1-\cos\theta)^2}{1-\cos^2\theta} = 1 + \dfrac{1-\cos\theta}{1+\cos\theta} = \dfrac{2}{1+\cos\theta}.
\]

令 $t = \cos\theta \in [-1, 1]$，则
\[
    f(t) = t + \dfrac{2}{1+t}, \quad t \in [-1, 1].
\]

求导得 $f'(t) = 1 - \dfrac{2}{(1+t)^2}$。令 $f'(t) = 0$，得 $(1+t)^2 = 2$，即 $t = \sqrt{2}-1$（舍去负根）。

当 $t = \sqrt{2}-1$ 时，$f(t) = (\sqrt{2}-1) + \dfrac{2}{\sqrt{2}} = (\sqrt{2}-1) + \sqrt{2} = 2\sqrt{2}-1$。

比较端点：$f(1) = 1 + 1 = 2$，$f(-1)$ 无定义（趋向 $+\infty$）。

所以最小值为 $2\sqrt{2}-1$。

\begin{realexam}[2025 宝山一模]
    已知平面向量 $\overrightarrow{a}$、$\overrightarrow{b}$ 满足：$|\overrightarrow{a}| = 1$，$|\overrightarrow{b}| = m \in (1, 2)$，且对任意的单位向量 $\overrightarrow{c}$ 满足 $|\overrightarrow{a} \cdot \overrightarrow{c}| + |\overrightarrow{b} \cdot \overrightarrow{c}| \leq \sqrt{6}$，则 $\overrightarrow{a} \cdot \overrightarrow{b}$ 的最大值为 $\_\_\_\_\_$（用含 $m$ 的式子表示）。
\end{realexam}

\textbf{解.} 建立坐标系，设 $\overrightarrow{a} = (1, 0)$，$\overrightarrow{b} = (m\cos\theta, m\sin\theta)$，其中 $\theta$ 为两向量夹角。

对任意单位向量 $\overrightarrow{c} = (\cos\varphi, \sin\varphi)$，有
\[
    |\overrightarrow{a} \cdot \overrightarrow{c}| + |\overrightarrow{b} \cdot \overrightarrow{c}| = |\cos\varphi| + |m\cos\theta\cos\varphi + m\sin\theta\sin\varphi| = |\cos\varphi| + m|\cos(\varphi-\theta)|.
\]

取 $\varphi = 0$，得 $1 + m|\cos\theta| \leq \sqrt{6}$；取 $\varphi = \theta$，得 $|\cos\theta| + m \leq \sqrt{6}$。

由第二个不等式，$|\cos\theta| \leq \sqrt{6} - m$。

因此
\[
    \overrightarrow{a} \cdot \overrightarrow{b} = m\cos\theta \leq m(\sqrt{6}-m) = m\sqrt{6} - m^2.
\]

当 $\cos\theta = \sqrt{6}-m$ 时取等号，此时需验证 $1 + m(\sqrt{6}-m) \leq \sqrt{6}$，即 $1 + m\sqrt{6} - m^2 \leq \sqrt{6}$，成立。

故 $\overrightarrow{a} \cdot \overrightarrow{b}$ 的最大值为 $m\sqrt{6} - m^2$。

\begin{example}[练习]
    在 $\triangle ABC$ 中，$AB = 3$，$AC = 4$，$\angle BAC = 60^{\circ}$，点 $D$ 满足 $\overrightarrow{BD} = 2\overrightarrow{DC}$，求 $\overrightarrow{AD} \cdot \overrightarrow{BC}$。
\end{example}

\textbf{解.} 由 $\overrightarrow{BD} = 2\overrightarrow{DC}$，得 $\overrightarrow{AD} - \overrightarrow{AB} = 2(\overrightarrow{AC} - \overrightarrow{AD})$，即 $3\overrightarrow{AD} = \overrightarrow{AB} + 2\overrightarrow{AC}$。

所以 $\overrightarrow{AD} = \dfrac{1}{3}\overrightarrow{AB} + \dfrac{2}{3}\overrightarrow{AC}$。

又 $\overrightarrow{BC} = \overrightarrow{AC} - \overrightarrow{AB}$，因此
\begin{align*}
    \overrightarrow{AD} \cdot \overrightarrow{BC} &= \left(\dfrac{1}{3}\overrightarrow{AB} + \dfrac{2}{3}\overrightarrow{AC}\right) \cdot (\overrightarrow{AC} - \overrightarrow{AB}) \\
    &= \dfrac{2}{3}|\overrightarrow{AC}|^2 - \dfrac{1}{3}|\overrightarrow{AB}|^2 - \dfrac{1}{3}\overrightarrow{AB} \cdot \overrightarrow{AC} \\
    &= \dfrac{2}{3} \times 16 - \dfrac{1}{3} \times 9 - \dfrac{1}{3} \times 3 \times 4 \times \dfrac{1}{2} \\
    &= \dfrac{32-9-6}{3} = \dfrac{17}{3}.
\end{align*}

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{基底法}：选择适当的基底向量（通常选已知模和夹角的向量），将所求向量用基底表示，再利用数量积运算求解。
    
    \item \textbf{坐标法}：建立平面直角坐标系，将向量问题转化为坐标运算，适用于有明确几何图形的问题。
    
    \item \textbf{几何意义法}：利用数量积的几何意义（投影）或向量加减法的几何意义（三角形法则、平行四边形法则）进行直观分析。
    
    \item \textbf{极化恒等式}：$\overrightarrow{a} \cdot \overrightarrow{b} = \dfrac{1}{4}(|\overrightarrow{a}+\overrightarrow{b}|^2 - |\overrightarrow{a}-\overrightarrow{b}|^2)$，适用于求数量积范围或最值问题。
    
    \item \textbf{建系策略}：遇到求最值或范围的问题，优先考虑建立坐标系，将向量条件转化为代数约束，再用函数或不等式工具求解。
\end{enumerate}
