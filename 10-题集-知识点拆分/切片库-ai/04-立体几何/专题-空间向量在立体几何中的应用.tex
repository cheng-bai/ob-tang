\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

空间向量法是解决立体几何问题的重要方法，通过建立空间直角坐标系，将几何元素用向量表示，可以将几何问题转化为代数运算，避免了复杂的几何推理。

\begin{definition}[直线的方向向量]
    如果向量 $\boldsymbol{v}$ 与直线 $l$ 平行（或共线），则称 $\boldsymbol{v}$ 为直线 $l$ 的\textbf{方向向量}。一条直线的方向向量有无数个，它们都是共线向量。
    
    过点 $P_0(x_0, y_0, z_0)$，方向向量为 $\boldsymbol{v} = (a, b, c)$ 的直线的参数方程为：
    \[
    \begin{cases}
    x = x_0 + at \\
    y = y_0 + bt \\
    z = z_0 + ct
    \end{cases} \quad (t \in \mathbb{R})
    \]
\end{definition}

\begin{definition}[平面的法向量]
    如果向量 $\boldsymbol{n}$ 与平面 $\alpha$ 垂直，则称 $\boldsymbol{n}$ 为平面 $\alpha$ 的\textbf{法向量}。一个平面的法向量有无数个，它们都是共线向量。
    
    设平面 $\alpha$ 过点 $P_0(x_0, y_0, z_0)$，法向量为 $\boldsymbol{n} = (A, B, C)$，则平面的点法式方程为：
    \[
    A(x - x_0) + B(y - y_0) + C(z - z_0) = 0
    \]
    即 $Ax + By + Cz + D = 0$，其中 $D = -(Ax_0 + By_0 + Cz_0)$。
\end{definition}

\begin{property}[法向量的求法]
    设平面 $\alpha$ 内有两个不共线向量 $\boldsymbol{a} = (x_1, y_1, z_1)$ 和 $\boldsymbol{b} = (x_2, y_2, z_2)$，则平面的法向量 $\boldsymbol{n}$ 与 $\boldsymbol{a}$、$\boldsymbol{b}$ 都垂直，可取：
    \[
    \boldsymbol{n} = \boldsymbol{a} \times \boldsymbol{b} = \begin{vmatrix} \boldsymbol{i} & \boldsymbol{j} & \boldsymbol{k} \\ x_1 & y_1 & z_1 \\ x_2 & y_2 & z_2 \end{vmatrix} = (y_1z_2 - z_1y_2, z_1x_2 - x_1z_2, x_1y_2 - y_1x_2)
    \]
\end{property}

\begin{property}[空间角的向量求法]
    设两直线的方向向量分别为 $\boldsymbol{v_1}$、$\boldsymbol{v_2}$，两平面的法向量分别为 $\boldsymbol{n_1}$、$\boldsymbol{n_2}$，直线方向向量为 $\boldsymbol{v}$，平面法向量为 $\boldsymbol{n}$：
    \begin{itemize}
        \item \textbf{线线角}：$\cos\theta = \dfrac{|\boldsymbol{v_1} \cdot \boldsymbol{v_2}|}{|\boldsymbol{v_1}||\boldsymbol{v_2}|}$，$\theta \in \left[0, \dfrac{\pi}{2}\right]$
        \item \textbf{线面角}：$\sin\theta = \dfrac{|\boldsymbol{v} \cdot \boldsymbol{n}|}{|\boldsymbol{v}||\boldsymbol{n}|}$，$\theta \in \left[0, \dfrac{\pi}{2}\right]$
        \item \textbf{二面角}：$\cos\theta = \dfrac{|\boldsymbol{n_1} \cdot \boldsymbol{n_2}|}{|\boldsymbol{n_1}||\boldsymbol{n_2}|}$ 或 $\cos\theta = -\dfrac{\boldsymbol{n_1} \cdot \boldsymbol{n_2}}{|\boldsymbol{n_1}||\boldsymbol{n_2}|}$，需结合图形判断
    \end{itemize}
\end{property}

\begin{property}[空间距离的向量求法]
    设点 $P(x_0, y_0, z_0)$，平面 $\alpha$ 的方程为 $Ax + By + Cz + D = 0$，则点 $P$ 到平面 $\alpha$ 的距离：
    \[
    d = \dfrac{|Ax_0 + By_0 + Cz_0 + D|}{\sqrt{A^2 + B^2 + C^2}}
    \]
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[建立坐标系求线面角]
    在正方体 $ABCD-A_1B_1C_1D_1$ 中，棱长为 $2$，$E$ 是 $BB_1$ 的中点。求直线 $AE$ 与平面 $A_1B_1C_1D_1$ 所成角的大小。
\end{example}

\textbf{解.} 建立空间直角坐标系，以 $D$ 为原点，$DA$ 为 $x$ 轴，$DC$ 为 $y$ 轴，$DD_1$ 为 $z$ 轴。

则 $A(2, 0, 0)$，$B(2, 2, 0)$，$B_1(2, 2, 2)$，$E$ 是 $BB_1$ 中点，所以 $E(2, 2, 1)$。

$\overrightarrow{AE} = (0, 2, 1)$。

平面 $A_1B_1C_1D_1$ 是上底面，其法向量可取 $\boldsymbol{n} = (0, 0, 1)$（$z$ 轴方向）。

设线面角为 $\theta$，则
\[
\sin\theta = \dfrac{|\overrightarrow{AE} \cdot \boldsymbol{n}|}{|\overrightarrow{AE}||\boldsymbol{n}|} = \dfrac{|0 \times 0 + 2 \times 0 + 1 \times 1|}{\sqrt{0 + 4 + 1} \times 1} = \dfrac{1}{\sqrt{5}} = \dfrac{\sqrt{5}}{5}
\]

所以 $\theta = \arcsin\dfrac{\sqrt{5}}{5}$。

\begin{example}[求二面角]
    在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是正方形，$PA \perp$ 平面 $ABCD$，$PA = AB = 2$。求二面角 $P-BC-A$ 的大小。
\end{example}

\textbf{解.} 建立空间直角坐标系，以 $A$ 为原点，$AB$ 为 $x$ 轴，$AD$ 为 $y$ 轴，$AP$ 为 $z$ 轴。

则 $A(0, 0, 0)$，$B(2, 0, 0)$，$C(2, 2, 0)$，$D(0, 2, 0)$，$P(0, 0, 2)$。

平面 $ABC$（即底面）的法向量 $\boldsymbol{n_1} = (0, 0, 1)$。

对于平面 $PBC$，$\overrightarrow{PB} = (2, 0, -2)$，$\overrightarrow{PC} = (2, 2, -2)$。

设平面 $PBC$ 的法向量 $\boldsymbol{n_2} = (x, y, z)$，则：
\[
\begin{cases}
\boldsymbol{n_2} \cdot \overrightarrow{PB} = 2x - 2z = 0 \\
\boldsymbol{n_2} \cdot \overrightarrow{PC} = 2x + 2y - 2z = 0
\end{cases}
\]

由第一式得 $x = z$，代入第二式得 $2x + 2y - 2x = 0$，即 $y = 0$。

取 $x = 1$，则 $\boldsymbol{n_2} = (1, 0, 1)$。

设二面角为 $\theta$，则
\[
\cos\theta = \dfrac{|\boldsymbol{n_1} \cdot \boldsymbol{n_2}|}{|\boldsymbol{n_1}||\boldsymbol{n_2}|} = \dfrac{|0 + 0 + 1|}{1 \times \sqrt{2}} = \dfrac{1}{\sqrt{2}} = \dfrac{\sqrt{2}}{2}
\]

所以 $\theta = \dfrac{\pi}{4}$（即 $45°$）。

\begin{realexam}[2025 上海奉贤二模]
    在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是菱形，$AC \cap BD = O$，$PA = PC = 5$，$PB = PD = 2\sqrt{5}$，$AC = 6$。
    \begin{enumerate}
        \item 求证：$PO \perp$ 平面 $ABCD$；
        \item 当 $\lambda = \dfrac{2}{3}$ 时，求 $AG$ 与底面 $ABCD$ 所成的线面角大小（其中 $G$ 是 $PC$ 上满足 $PG = \lambda PC$ 的点）。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 证明：因为 $ABCD$ 是菱形，$O$ 是对角线交点，所以 $O$ 是 $AC$ 和 $BD$ 的中点。

在 $\triangle PAC$ 中，$PA = PC = 5$，所以 $\triangle PAC$ 是等腰三角形，$PO \perp AC$。

在 $\triangle PBD$ 中，$PB = PD = 2\sqrt{5}$，所以 $\triangle PBD$ 是等腰三角形，$PO \perp BD$。

因为 $AC \cap BD = O$，$AC, BD \subset$ 平面 $ABCD$，所以 $PO \perp$ 平面 $ABCD$。

(2) 以 $O$ 为坐标原点，$OA$ 为 $x$ 轴，$OB$ 为 $y$ 轴，$OP$ 为 $z$ 轴建立空间直角坐标系。

由 $AC = 6$，得 $OA = 3$。在 $\text{Rt}\triangle POA$ 中，$PO = \sqrt{PA^2 - OA^2} = \sqrt{25 - 9} = 4$。

在 $\text{Rt}\triangle POB$ 中，$OB = \sqrt{PB^2 - PO^2} = \sqrt{20 - 16} = 2$。

所以 $A(3, 0, 0)$，$C(-3, 0, 0)$，$B(0, 2, 0)$，$D(0, -2, 0)$，$P(0, 0, 4)$。

当 $\lambda = \dfrac{2}{3}$ 时，$\overrightarrow{PG} = \dfrac{2}{3}\overrightarrow{PC} = \dfrac{2}{3}(-3, 0, -4) = (-2, 0, -\dfrac{8}{3})$。

$G = P + \overrightarrow{PG} = (0, 0, 4) + (-2, 0, -\dfrac{8}{3}) = (-2, 0, \dfrac{4}{3})$。

$\overrightarrow{AG} = (-2 - 3, 0 - 0, \dfrac{4}{3} - 0) = (-5, 0, \dfrac{4}{3})$。

底面 $ABCD$ 的法向量 $\boldsymbol{n} = (0, 0, 1)$。

设线面角为 $\theta$，则
\[
\sin\theta = \dfrac{|\overrightarrow{AG} \cdot \boldsymbol{n}|}{|\overrightarrow{AG}||\boldsymbol{n}|} = \dfrac{|\dfrac{4}{3}|}{\sqrt{25 + 0 + \dfrac{16}{9}} \times 1} = \dfrac{\dfrac{4}{3}}{\sqrt{\dfrac{225 + 16}{9}}} = \dfrac{\dfrac{4}{3}}{\dfrac{\sqrt{241}}{3}} = \dfrac{4}{\sqrt{241}} = \dfrac{4\sqrt{241}}{241}
\]

所以 $\theta = \arcsin\dfrac{4\sqrt{241}}{241}$。

\begin{realexam}[2024 上海二模]
    在三棱锥 $P-ABC$ 中，$PA \perp$ 平面 $ABC$，$PA = AB = AC = 2$，$\angle BAC = 90°$。
    \begin{enumerate}
        \item 建立适当的空间直角坐标系，写出各顶点坐标；
        \item 求平面 $PBC$ 与平面 $ABC$ 所成二面角的大小；
        \item 求点 $A$ 到平面 $PBC$ 的距离。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 以 $A$ 为原点，$AB$ 为 $x$ 轴，$AC$ 为 $y$ 轴，$AP$ 为 $z$ 轴建立空间直角坐标系。

则 $A(0, 0, 0)$，$B(2, 0, 0)$，$C(0, 2, 0)$，$P(0, 0, 2)$。

(2) 平面 $ABC$ 的法向量 $\boldsymbol{n_1} = (0, 0, 1)$。

对于平面 $PBC$，$\overrightarrow{PB} = (2, 0, -2)$，$\overrightarrow{PC} = (0, 2, -2)$。

设平面 $PBC$ 的法向量 $\boldsymbol{n_2} = (x, y, z)$，则：
\[
\begin{cases}
2x - 2z = 0 \\
2y - 2z = 0
\end{cases}
\]

得 $x = z$，$y = z$。取 $z = 1$，则 $\boldsymbol{n_2} = (1, 1, 1)$。

$\cos\theta = \dfrac{|\boldsymbol{n_1} \cdot \boldsymbol{n_2}|}{|\boldsymbol{n_1}||\boldsymbol{n_2}|} = \dfrac{1}{\sqrt{3}} = \dfrac{\sqrt{3}}{3}$

所以二面角为 $\arccos\dfrac{\sqrt{3}}{3}$。

(3) 点 $A(0, 0, 0)$ 到平面 $PBC$ 的距离。

平面 $PBC$ 过点 $P(0, 0, 2)$，法向量 $\boldsymbol{n_2} = (1, 1, 1)$。

平面方程为 $1(x - 0) + 1(y - 0) + 1(z - 2) = 0$，即 $x + y + z - 2 = 0$。

点 $A$ 到平面的距离 $d = \dfrac{|0 + 0 + 0 - 2|}{\sqrt{1 + 1 + 1}} = \dfrac{2}{\sqrt{3}} = \dfrac{2\sqrt{3}}{3}$。

\begin{example}[练习]
    在长方体 $ABCD-A_1B_1C_1D_1$ 中，$AB = 2$，$BC = 3$，$AA_1 = 4$。求：
    \begin{enumerate}
        \item 异面直线 $A_1B$ 与 $B_1C$ 所成角的大小；
        \item 直线 $A_1B$ 与平面 $BCC_1B_1$ 所成角的大小。
    \end{enumerate}
\end{example}

\textbf{解.} 建立空间直角坐标系，以 $D$ 为原点，$DA$ 为 $x$ 轴，$DC$ 为 $y$ 轴，$DD_1$ 为 $z$ 轴。

则 $A_1(3, 0, 4)$，$B(3, 2, 0)$，$B_1(3, 2, 4)$，$C(0, 2, 0)$。

(1) $\overrightarrow{A_1B} = (0, 2, -4)$，$\overrightarrow{B_1C} = (-3, 0, -4)$。

$\overrightarrow{A_1B} \cdot \overrightarrow{B_1C} = 0 \times (-3) + 2 \times 0 + (-4) \times (-4) = 16$

$|\overrightarrow{A_1B}| = \sqrt{0 + 4 + 16} = \sqrt{20} = 2\sqrt{5}$

$|\overrightarrow{B_1C}| = \sqrt{9 + 0 + 16} = \sqrt{25} = 5$

$\cos\theta = \dfrac{16}{2\sqrt{5} \times 5} = \dfrac{16}{10\sqrt{5}} = \dfrac{8\sqrt{5}}{25}$

所以异面直线所成角为 $\arccos\dfrac{8\sqrt{5}}{25}$。

(2) 平面 $BCC_1B_1$ 是侧面，其法向量 $\boldsymbol{n} = (1, 0, 0)$（$x$ 轴方向，即垂直于该侧面）。

$\sin\phi = \dfrac{|\overrightarrow{A_1B} \cdot \boldsymbol{n}|}{|\overrightarrow{A_1B}||\boldsymbol{n}|} = \dfrac{|0|}{2\sqrt{5} \times 1} = 0$

所以 $\phi = 0$，即直线 $A_1B$ 与平面 $BCC_1B_1$ 平行。

验证：$\overrightarrow{A_1B} = (0, 2, -4)$，平面 $BCC_1B_1$ 内的向量 $\overrightarrow{BC} = (-3, 0, 0)$，$\overrightarrow{BB_1} = (0, 0, 4)$。

$\overrightarrow{A_1B}$ 与平面不垂直，且 $\overrightarrow{A_1B} \cdot \boldsymbol{n} = 0$ 说明与法向量垂直，即与平面平行。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{建立坐标系的原则}：
    \begin{itemize}
        \item 尽可能利用几何体中现成的垂直关系；
        \item 将原点设在便于计算的位置（如顶点、底面中心）；
        \item 使尽可能多的点落在坐标轴上或坐标平面上。
    \end{itemize}
    
    \item \textbf{法向量的求法}：
    \begin{itemize}
        \item 找平面内两个不共线向量，用叉乘求法向量；
        \item 利用平面与坐标轴的垂直关系直接写法向量；
        \item 设方程法：设法向量 $(A, B, C)$，利用垂直条件列方程求解。
    \end{itemize}
    
    \item \textbf{空间角的计算要点}：
    \begin{itemize}
        \item 线线角：用方向向量夹角公式，取锐角；
        \item 线面角：用方向向量与法向量夹角的余角，$\sin\theta = |\cos\langle\boldsymbol{v}, \boldsymbol{n}\rangle|$；
        \item 二面角：用法向量夹角，注意判断是锐角还是钝角。
    \end{itemize}
    
    \item \textbf{点到平面距离}：
    \begin{itemize}
        \item 直接套用点法式距离公式；
        \item 或转化为向量在法向量上的投影长度。
    \end{itemize}
    
    \item \textbf{注意事项}：
    \begin{itemize}
        \item 计算过程中要注意符号，特别是二面角的判断；
        \item 最终结果要化为最简形式；
        \item 对于特殊角，要给出具体角度值（如 $30°$、$45°$、$60°$ 等）。
    \end{itemize}
\end{enumerate}
