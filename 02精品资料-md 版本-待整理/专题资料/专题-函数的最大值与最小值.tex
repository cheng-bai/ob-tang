\begin{center}
    {\LARGE\bfseries 专题：函数的最大值与最小值}
\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

函数的最值问题是函数应用的核心内容，涉及求函数在特定区间或整个定义域上的最大值和最小值，是解决优化问题、不等式证明等的重要工具。

\begin{definition}[最大值与最小值]
    设函数 $y = f(x)$ 的定义域为 $D$，区间 $I \subseteq D$：
    \begin{itemize}
        \item 若存在 $x_0 \in I$，使得对任意 $x \in I$，都有 $f(x) \leq f(x_0)$，则称 $f(x_0)$ 为函数在区间 $I$ 上的\textbf{最大值}，记作 $f(x)_{\max}$ 或 $\max\limits_{x \in I} f(x)$；
        \item 若存在 $x_0 \in I$，使得对任意 $x \in I$，都有 $f(x) \geq f(x_0)$，则称 $f(x_0)$ 为函数在区间 $I$ 上的\textbf{最小值}，记作 $f(x)_{\min}$ 或 $\min\limits_{x \in I} f(x)$。
    \end{itemize}
\end{definition}

\begin{property}[求最值的基本方法]
    \begin{enumerate}
        \item \textbf{配方法}：适用于二次函数，通过配方确定顶点坐标；
        \item \textbf{判别式法}：适用于可化为关于 $x$ 的二次方程的函数；
        \item \textbf{单调性法}：利用函数的单调性确定端点值或极值；
        \item \textbf{导数法}：求导找驻点，比较驻点和端点的函数值；
        \item \textbf{基本不等式法}：利用均值不等式求最值；
        \item \textbf{换元法}：通过换元转化为已知最值的函数；
        \item \textbf{数形结合法}：利用几何意义求最值。
    \end{enumerate}
\end{property}

\begin{property}[闭区间上连续函数的最值]
    若函数 $f(x)$ 在闭区间 $[a, b]$ 上连续，则 $f(x)$ 在 $[a, b]$ 上一定存在最大值和最小值。求法：
    \begin{enumerate}
        \item 求 $f(x)$ 在 $(a, b)$ 内的所有驻点（$f'(x) = 0$ 的点）和不可导点；
        \item 计算这些点及区间端点 $a, b$ 处的函数值；
        \item 比较上述函数值，最大者为最大值，最小者为最小值。
    \end{enumerate}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[利用基本不等式求最值]
    已知 $x > 0, y > 0$，且 $xy = 1$，求 $x + 2y$ 的最小值。
\end{example}

\textbf{解.} 由基本不等式：
\[
    x + 2y \geq 2\sqrt{x \cdot 2y} = 2\sqrt{2xy} = 2\sqrt{2}
\]

当且仅当 $x = 2y$ 且 $xy = 1$ 时取等号，即 $x = \sqrt{2}, y = \dfrac{\sqrt{2}}{2}$。

因此 $x + 2y$ 的最小值为 $2\sqrt{2}$。

\begin{example}[二次函数在闭区间上的最值]
    求函数 $f(x) = x^2 - 2ax + 3$ 在区间 $[0, 2]$ 上的最大值和最小值。
\end{example}

\textbf{解.} 配方得 $f(x) = (x-a)^2 + 3 - a^2$，对称轴为 $x = a$。

\textbf{求最小值：}
\begin{itemize}
    \item 当 $a < 0$ 时，$f(x)$ 在 $[0, 2]$ 上递增，$f(x)_{\min} = f(0) = 3$；
    \item 当 $0 \leq a \leq 2$ 时，$f(x)_{\min} = f(a) = 3 - a^2$；
    \item 当 $a > 2$ 时，$f(x)$ 在 $[0, 2]$ 上递减，$f(x)_{\min} = f(2) = 7 - 4a$。
\end{itemize}

\textbf{求最大值：}
\begin{itemize}
    \item 当 $a \leq 1$ 时，$f(x)_{\max} = f(2) = 7 - 4a$；
    \item 当 $a > 1$ 时，$f(x)_{\max} = f(0) = 3$。
\end{itemize}

\begin{realexam}[2025 上海高三二模]
    已知 $a > 0, b > 0$ 且 $a + b = 2$，求 $ab$ 的最大值。
\end{realexam}

\textbf{解.} 由基本不等式：
\[
    ab \leq \left(\frac{a+b}{2}\right)^2 = \left(\frac{2}{2}\right)^2 = 1
\]

当且仅当 $a = b = 1$ 时取等号。

因此 $ab$ 的最大值为 $1$。

\begin{realexam}[2025 上海高三二模]
    已知 $a > b$，$ab = 1$，求 $\dfrac{a^2 + b^2}{a-b}$ 的最小值。
\end{realexam}

\textbf{解.} 由 $ab = 1$，得：
\[
    \frac{a^2 + b^2}{a-b} = \frac{(a-b)^2 + 2ab}{a-b} = \frac{(a-b)^2 + 2}{a-b} = (a-b) + \frac{2}{a-b}
\]

设 $t = a - b > 0$，则原式 $= t + \dfrac{2}{t}$。

由基本不等式：
\[
    t + \frac{2}{t} \geq 2\sqrt{t \cdot \frac{2}{t}} = 2\sqrt{2}
\]

当且仅当 $t = \dfrac{2}{t}$，即 $t = \sqrt{2}$ 时取等号。

因此最小值为 $2\sqrt{2}$。

\begin{example}[利用导数求最值]
    求函数 $f(x) = x^3 - 3x^2 + 2$ 在区间 $[-1, 3]$ 上的最大值和最小值。
\end{example}

\textbf{解.} 求导得 $f'(x) = 3x^2 - 6x = 3x(x-2)$。

令 $f'(x) = 0$，得驻点 $x_1 = 0, x_2 = 2$。

计算关键点函数值：
\[
    f(-1) = (-1)^3 - 3(-1)^2 + 2 = -1 - 3 + 2 = -2
\]
\[
    f(0) = 0 - 0 + 2 = 2
\]
\[
    f(2) = 8 - 12 + 2 = -2
\]
\[
    f(3) = 27 - 27 + 2 = 2
\]

比较得：最大值为 $2$（在 $x = 0$ 和 $x = 3$ 处取得），最小值为 $-2$（在 $x = -1$ 和 $x = 2$ 处取得）。

\begin{realexam}[2025 上海高三二模]
    函数 $y = 2\sin x - 3\cos x$ 的值域是 $\_\_\_\_\_\_$。
\end{realexam}

\textbf{解.} 利用辅助角公式：
\[
    y = 2\sin x - 3\cos x = \sqrt{2^2 + (-3)^2} \sin(x + \varphi) = \sqrt{13} \sin(x + \varphi)
\]

其中 $\tan\varphi = -\dfrac{3}{2}$。

由于 $\sin(x+\varphi) \in [-1, 1]$，所以 $y \in [-\sqrt{13}, \sqrt{13}]$。

因此值域为 $[-\sqrt{13}, \sqrt{13}]$。

\begin{example}[分式函数的最值]
    求函数 $f(x) = \dfrac{x^2 + x + 1}{x^2 + 1}$ 的最大值和最小值。
\end{example}

\textbf{解.} 设 $y = \dfrac{x^2 + x + 1}{x^2 + 1}$，则：
\[
    y(x^2 + 1) = x^2 + x + 1
\]
\[
    (y-1)x^2 - x + (y-1) = 0
\]

当 $y = 1$ 时，$x = 0$，满足条件。

当 $y \neq 1$ 时，由判别式 $\Delta \geq 0$：
\[
    \Delta = 1 - 4(y-1)^2 \geq 0
\]
\[
    (y-1)^2 \leq \frac{1}{4}
\]
\[
    -\frac{1}{2} \leq y-1 \leq \frac{1}{2}
\]
\[
    \frac{1}{2} \leq y \leq \frac{3}{2}
\]

因此最大值为 $\dfrac{3}{2}$，最小值为 $\dfrac{1}{2}$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{基本不等式法要点}：
    \begin{itemize}
        \item 一正：各项必须为正数；
        \item 二定：和或积为定值；
        \item 三相等：等号成立条件必须能取到。
    \end{itemize}
    \item \textbf{二次函数最值}：注意对称轴与区间的位置关系，分类讨论是常用策略。
    \item \textbf{导数法求最值}：求闭区间上最值时，必须比较驻点、不可导点和端点的函数值，不可遗漏。
    \item \textbf{三角函数最值}：利用辅助角公式 $a\sin x + b\cos x = \sqrt{a^2+b^2}\sin(x+\varphi)$，值域为 $[-\sqrt{a^2+b^2}, \sqrt{a^2+b^2}]$。
    \item \textbf{判别式法}：适用于可化为关于 $x$ 的二次方程的函数，注意讨论二次项系数是否为零。
    \item \textbf{换元法}：通过换元将复杂函数转化为简单函数求最值，注意新变量的取值范围。
    \item \textbf{实际问题的最值}：建立目标函数后，需考虑实际意义对定义域的限制，验证等号成立条件是否符合实际。
\end{enumerate}
