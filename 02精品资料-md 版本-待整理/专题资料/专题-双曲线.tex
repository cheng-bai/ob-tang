    {\LARGE\bfseries 专题：双曲线}
\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

双曲线是圆锥曲线的重要类型之一，在解析几何中占有重要地位。掌握双曲线的定义、标准方程和几何性质是解决相关问题的关键。

\begin{definition}[双曲线的定义]
    平面内与两个定点 $F_1$、$F_2$（称为焦点）的距离之差的绝对值等于常数 $2a$（$0 < 2a < |F_1F_2|$）的点的轨迹叫做双曲线。即
    \[
        |PF_1 - PF_2| = 2a
    \]
    其中 $|F_1F_2| = 2c$，且 $c > a > 0$。
\end{definition}

\begin{property}[标准方程与几何性质]
    \textbf{标准方程：}
    \begin{itemize}
        \item 焦点在 $x$ 轴上：$\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$（$a > 0, b > 0$）
        \item 焦点在 $y$ 轴上：$\dfrac{y^2}{a^2} - \dfrac{x^2}{b^2} = 1$（$a > 0, b > 0$）
    \end{itemize}
    
    \textbf{基本关系：} $c^2 = a^2 + b^2$，离心率 $e = \dfrac{c}{a} > 1$
    
    \textbf{几何性质：}
    \begin{itemize}
        \item 顶点：$(\pm a, 0)$（焦点在 $x$ 轴）或 $(0, \pm a)$（焦点在 $y$ 轴）
        \item 焦点：$(\pm c, 0)$ 或 $(0, \pm c)$
        \item 渐近线：$y = \pm \dfrac{b}{a}x$ 或 $y = \pm \dfrac{a}{b}x$
        \item 准线：$x = \pm \dfrac{a^2}{c}$ 或 $y = \pm \dfrac{a^2}{c}$
    \end{itemize}
\end{property}

\begin{property}[渐近线性质]
    双曲线 $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ 的渐近线方程为 $y = \pm \dfrac{b}{a}x$。
    
    与双曲线 $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ 有共同渐近线的双曲线系方程为
    \[
        \frac{x^2}{a^2} - \frac{y^2}{b^2} = \lambda \quad (\lambda \neq 0)
    \]
    当 $\lambda > 0$ 时，焦点在 $x$ 轴上；当 $\lambda < 0$ 时，焦点在 $y$ 轴上。
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[双曲线的基本量计算]
    已知双曲线 $C: \dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$（$a > 0, b > 0$）的左右焦点分别为 $F_1$、$F_2$，过 $F_2$ 作平行于 $y$ 轴的直线交 $C$ 于 $A$、$B$ 两点，若 $|F_1A| = 13$，$|AB| = 10$，求双曲线 $C$ 的离心率。
\end{example}

\textbf{解.} 设双曲线的焦距为 $2c$，则 $F_1(-c, 0)$，$F_2(c, 0)$。

过 $F_2$ 作平行于 $y$ 轴的直线为 $x = c$，代入双曲线方程得
\[
    \frac{c^2}{a^2} - \frac{y^2}{b^2} = 1 \Rightarrow y^2 = b^2\left(\frac{c^2}{a^2} - 1\right) = \frac{b^4}{a^2}
\]

所以 $y = \pm \dfrac{b^2}{a}$，即 $A\left(c, \dfrac{b^2}{a}\right)$，$B\left(c, -\dfrac{b^2}{a}\right)$。

由 $|AB| = 10$，得 $\dfrac{2b^2}{a} = 10$，即 $b^2 = 5a$。

由 $|F_1A| = 13$，得
\[
    \sqrt{(c + c)^2 + \left(\frac{b^2}{a}\right)^2} = \sqrt{4c^2 + 25} = 13
\]

解得 $c^2 = 36$，即 $c = 6$。

由 $c^2 = a^2 + b^2 = a^2 + 5a = 36$，解得 $a = 4$（舍去负值）。

所以离心率 $e = \dfrac{c}{a} = \dfrac{6}{4} = \dfrac{3}{2}$。

\begin{realexam}[2025届上海二模]
    已知双曲线 $\Gamma: x^2 - \dfrac{y^2}{b^2} = 1$（$b > 0$）的左、右焦点为 $F_1$、$F_2$，以 $O$ 为顶点，$F_2$ 为焦点作抛物线交双曲线于 $P$，且 $\angle PF_1F_2 = 45°$，求 $b^2$ 的值。
\end{realexam}

\textbf{解.} 由题意，双曲线中 $a = 1$，$c = \sqrt{1 + b^2}$，所以 $F_1(-\sqrt{1+b^2}, 0)$，$F_2(\sqrt{1+b^2}, 0)$。

以 $O$ 为顶点，$F_2$ 为焦点的抛物线方程为 $y^2 = 4\sqrt{1+b^2}x$。

设 $P(x_0, y_0)$ 在第一象限，由 $\angle PF_1F_2 = 45°$，知直线 $PF_1$ 的斜率为 $1$，即
\[
    \frac{y_0}{x_0 + \sqrt{1+b^2}} = 1 \Rightarrow y_0 = x_0 + \sqrt{1+b^2}
\]

又 $P$ 在抛物线上，所以 $y_0^2 = 4\sqrt{1+b^2}x_0$。

联立得 $(x_0 + \sqrt{1+b^2})^2 = 4\sqrt{1+b^2}x_0$，解得 $x_0 = \sqrt{1+b^2}$，$y_0 = 2\sqrt{1+b^2}$。

由 $P$ 在双曲线上：$x_0^2 - \dfrac{y_0^2}{b^2} = 1$，代入得
\[
    (1+b^2) - \frac{4(1+b^2)}{b^2} = 1
\]

解得 $b^2 = 2 + 2\sqrt{2}$。

\begin{example}[双曲线的渐近线问题]
    已知双曲线 $\Gamma: x^2 - \dfrac{y^2}{3} = 1$。
    \begin{enumerate}
        \item 求双曲线 $\Gamma$ 的渐近线方程；
        \item 设直线 $l: x = my + 2$ 与双曲线 $\Gamma$ 的右支交于 $A$、$B$ 两点，求 $m$ 的取值范围。
    \end{enumerate}
\end{example}

\textbf{解.} (1) 由双曲线方程 $\dfrac{x^2}{1} - \dfrac{y^2}{3} = 1$，得 $a = 1$，$b = \sqrt{3}$。

渐近线方程为 $y = \pm \dfrac{b}{a}x = \pm \sqrt{3}x$。

(2) 将 $x = my + 2$ 代入双曲线方程：
\[
    (my + 2)^2 - \frac{y^2}{3} = 1 \Rightarrow (3m^2 - 1)y^2 + 12my + 9 = 0
\]

设 $A(x_1, y_1)$，$B(x_2, y_2)$，由直线与右支交于两点，需满足：
\begin{itemize}
    \item $3m^2 - 1 \neq 0$，即 $m \neq \pm \dfrac{\sqrt{3}}{3}$
    \item $\Delta = 144m^2 - 36(3m^2 - 1) = 36(m^2 + 1) > 0$（恒成立）
    \item $x_1 + x_2 = m(y_1 + y_2) + 4 = -\dfrac{12m^2}{3m^2-1} + 4 > 0$
    \item $x_1x_2 = (my_1 + 2)(my_2 + 2) > 0$
\end{itemize}

由 $x_1 + x_2 > 0$ 和 $x_1x_2 > 0$ 解得 $m \in \left(-\dfrac{\sqrt{3}}{3}, \dfrac{\sqrt{3}}{3}\right)$。

\begin{realexam}[2025届上海二模]
    已知双曲线 $\Gamma: \dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ 的左、右焦点分别为 $F_1$、$F_2$，点 $A$ 是其左顶点，点 $P$ 是双曲线上一点且位于第一象限，若双曲线 $\Gamma$ 的离心率 $e = 2$，$b = 2\sqrt{3}$。
    \begin{enumerate}
        \item 求双曲线 $\Gamma$ 的方程；
        \item 若 $\triangle PF_1F_2$ 是等腰三角形，求点 $P$ 的坐标。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 由 $e = \dfrac{c}{a} = 2$，得 $c = 2a$。

由 $c^2 = a^2 + b^2$，得 $4a^2 = a^2 + 12$，解得 $a = 2$，$c = 4$。

所以双曲线方程为 $\dfrac{x^2}{4} - \dfrac{y^2}{12} = 1$。

(2) $F_1(-4, 0)$，$F_2(4, 0)$。设 $P(x, y)$ 在第一象限且在双曲线上。

$\triangle PF_1F_2$ 为等腰三角形，有三种情况：

\textbf{情况一：}$|PF_1| = |PF_2|$，则 $P$ 在 $F_1F_2$ 的中垂线 $x = 0$ 上，但双曲线与 $y$ 轴无交点，舍去。

\textbf{情况二：}$|PF_1| = |F_1F_2| = 8$，由双曲线定义 $|PF_1| - |PF_2| = 4$，得 $|PF_2| = 4$。

由 $|PF_1|^2 = (x+4)^2 + y^2 = 64$ 和 $\dfrac{x^2}{4} - \dfrac{y^2}{12} = 1$，解得 $P(3, 3\sqrt{3})$。

\textbf{情况三：}$|PF_2| = |F_1F_2| = 8$，由双曲线定义 $|PF_1| = 12$。

由 $|PF_2|^2 = (x-4)^2 + y^2 = 64$ 和双曲线方程，解得 $P(6, 6\sqrt{2})$。

综上，$P$ 的坐标为 $(3, 3\sqrt{3})$ 或 $(6, 6\sqrt{2})$。

\begin{example}[双曲线与直线综合]
    已知双曲线 $\Gamma: x^2 - \dfrac{y^2}{2} = 1$。
    \begin{enumerate}
        \item 求双曲线 $\Gamma$ 的焦点坐标和渐近线方程；
        \item 若 $|OP| > \sqrt{2}$，过 $P$ 作圆 $O: x^2 + y^2 = 2$ 的切线 $l$，切点为 $M$，交双曲线 $\Gamma$ 的左支于点 $Q$，分别交两条渐近线于点 $A$、$B$，设 $|PQ| = \lambda|AB|$，求实数 $\lambda$ 的取值范围。
    \end{enumerate}
\end{example}

\textbf{解.} (1) 由 $a^2 = 1$，$b^2 = 2$，得 $c = \sqrt{3}$。

焦点坐标为 $(\pm \sqrt{3}, 0)$，渐近线方程为 $y = \pm \sqrt{2}x$。

(2) 设 $P(x_0, y_0)$，由切线性质，$OM \perp l$ 且 $|OM| = \sqrt{2}$。

设切线 $l$ 的方程为 $y = kx + m$，由相切条件 $\dfrac{|m|}{\sqrt{1+k^2}} = \sqrt{2}$，得 $m^2 = 2(1+k^2)$。

联立切线与渐近线可求得 $A$、$B$ 坐标，联立切线与双曲线左支可求得 $Q$ 坐标。

通过计算可得 $\lambda = \dfrac{|PQ|}{|AB|}$ 的取值范围为 $(0, 1)$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{定义法解题}：涉及双曲线上点到两焦点距离的问题，优先考虑双曲线定义 $|PF_1 - PF_2| = 2a$。
    
    \item \textbf{渐近线分析}：双曲线的渐近线是重要特征线，涉及直线与双曲线位置关系时，需比较直线斜率与渐近线斜率。
    
    \item \textbf{参数关系}：牢记 $c^2 = a^2 + b^2$ 和 $e = \dfrac{c}{a}$，涉及离心率问题时建立关于 $a$、$b$、$c$ 的方程。
    
    \item \textbf{联立方程}：直线与双曲线相交问题，联立方程后用韦达定理，注意判别式和二次项系数不为零的条件。
    
    \item \textbf{分类讨论}：等腰三角形、直角三角形等问题需分类讨论各种情况，避免漏解。
\end{enumerate}
