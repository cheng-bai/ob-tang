\begin{center}
    {\LARGE\bfseries 专题：函数的单调性}
\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

函数的单调性是描述函数值随自变量增大（或减小）而变化趋势的重要性质，是研究函数图像、最值、不等式及方程的基础工具。

\begin{definition}[单调递增函数]
    设函数 $y = f(x)$ 的定义域为 $D$，区间 $I \subseteq D$。若对任意的 $x_1, x_2 \in I$，当 $x_1 < x_2$ 时，都有
    \[
        f(x_1) < f(x_2),
    \]
    则称函数 $y = f(x)$ 在区间 $I$ 上\textbf{严格单调递增}（简称单调递增）。
\end{definition}

类似地，若当 $x_1 < x_2$ 时，都有 $f(x_1) > f(x_2)$，则称函数在区间 $I$ 上\textbf{严格单调递减}（简称单调递减）。单调递增区间与单调递减区间统称为函数的\textbf{单调区间}。

\begin{property}[定义法判定]
    判断函数 $f(x)$ 在区间 $I$ 上的单调性，最基本的方法是\textbf{定义法}：
    \begin{enumerate}
        \item 在区间 $I$ 内任取 $x_1 < x_2$；
        \item 作差 $f(x_1) - f(x_2)$，判断其符号；
        \item 若 $f(x_1) - f(x_2) < 0$，则 $f(x)$ 在 $I$ 上单调递增；若 $f(x_1) - f(x_2) > 0$，则单调递减。
    \end{enumerate}
\end{property}

\begin{property}[导数法判定]
    设函数 $y = f(x)$ 在区间 $I$ 上可导：
    \begin{itemize}
        \item 若 $f'(x) > 0$ 在 $I$ 上恒成立，则 $f(x)$ 在 $I$ 上单调递增；
        \item 若 $f'(x) < 0$ 在 $I$ 上恒成立，则 $f(x)$ 在 $I$ 上单调递减。
    \end{itemize}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[利用导数求单调区间]
    设函数 $f(x) = x^3 + x^2 - 8x + 7$，求：
    \begin{enumerate}
        \item 函数 $y = f(x)$ 的单调区间；
        \item 函数 $y = f(x)$ 在区间 $[-3, 3]$ 上的最大值和最小值。
    \end{enumerate}
\end{example}

\textbf{解.} (1) 求导得 $f'(x) = 3x^2 + 2x - 8 = (3x - 4)(x + 2)$。

令 $f'(x) = 0$，解得驻点 $x_1 = -2$，$x_2 = \dfrac{4}{3}$。

分析导数符号：
\begin{itemize}
    \item 当 $x < -2$ 或 $x > \dfrac{4}{3}$ 时，$f'(x) > 0$，函数严格增；
    \item 当 $-2 < x < \dfrac{4}{3}$ 时，$f'(x) < 0$，函数严格减。
\end{itemize}

因此，函数 $f(x)$ 的单调增区间为 $(-\infty, -2)$ 和 $\left( \dfrac{4}{3}, +\infty \right)$，单调减区间为 $\left( -2, \dfrac{4}{3} \right)$。

(2) 由 (1) 可知，$f(x)$ 在 $[-3, -2]$ 和 $\left[ \dfrac{4}{3}, 3 \right]$ 上严格增，在 $\left[ -2, \dfrac{4}{3} \right]$ 上严格减。计算关键点函数值：
\[
    f(-3) = 13, \quad f(-2) = 19, \quad f\left( \frac{4}{3} \right) = \frac{13}{27}, \quad f(3) = 19.
\]

所以，最大值为 $19$（在 $x = -2$ 和 $x = 3$ 处取得），最小值为 $\dfrac{13}{27}$（在 $x = \dfrac{4}{3}$ 处取得）。

\begin{realexam}[2013 上海秋考]
    已知函数 $f(x) = 2\sin(\omega x)$，其中常数 $\omega > 0$。
    \begin{enumerate}
        \item 若 $y = f(x)$ 在 $\left[ -\dfrac{\pi}{4}, \dfrac{2\pi}{3} \right]$ 上单调递增，求 $\omega$ 的取值范围；
        \item 令 $\omega = 2$，将函数 $y = f(x)$ 的图象向左平移 $\dfrac{\pi}{6}$ 个单位，再向上平移 $1$ 个单位，得到函数 $y = g(x)$ 的图象。区间 $[a, b]$（$a, b \in \mathbb{R}$ 且 $a < b$）满足：$y = g(x)$ 在 $[a, b]$ 上至少含有 $30$ 个零点，在所有满足上述条件的 $[a, b]$ 中，求 $b - a$ 的最小值。
    \end{enumerate}
\end{realexam}

\textbf{解.} (1) 因为 $\omega > 0$，$y = f(x) = 2\sin(\omega x)$ 在 $\left[ -\dfrac{\pi}{4}, \dfrac{2\pi}{3} \right]$ 上单调递增，需满足
\[
    \begin{cases}
        -\dfrac{\pi}{4}\omega \ge -\dfrac{\pi}{2} \\[6pt]
        \dfrac{2\pi}{3}\omega \le \dfrac{\pi}{2}
    \end{cases}
\]
解得 $0 < \omega \le \dfrac{3}{4}$。所以 $\omega$ 的取值范围为 $\left( 0, \dfrac{3}{4} \right]$。

(2) 令 $\omega = 2$，则 $f(x) = 2\sin(2x)$。向左平移 $\dfrac{\pi}{6}$ 个单位，再向上平移 $1$ 个单位，得
\[
    g(x) = 2\sin\left( 2x + \frac{\pi}{3} \right) + 1.
\]

令 $g(x) = 0$，得 $\sin\left( 2x + \dfrac{\pi}{3} \right) = -\dfrac{1}{2}$，解得
\[
    x = k\pi + \frac{5\pi}{12} \quad \text{或} \quad x = k\pi + \frac{3\pi}{4}, \quad k \in \mathbb{Z}.
\]

相邻两个零点之间的距离为 $\dfrac{\pi}{3}$ 或 $\dfrac{2\pi}{3}$。要使 $b - a$ 最小且包含至少 $30$ 个零点，取间隔最紧凑的排列，可得 $b - a$ 的最小值为 $\dfrac{43\pi}{3}$。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{定义域优先}：研究单调性前，必须先确定函数的定义域，单调区间一定是定义域的子集。
    \item \textbf{定义法与导数法}：不含参数的具体函数，优先用导数法求单调区间；抽象函数或需要严格证明的场合，可用定义法。
    \item \textbf{三角函数单调性}：对于 $y = A\sin(\omega x + \varphi)$（$\omega > 0$），将其整体代入正弦函数的单调区间求解；注意参数 $\omega$ 对周期和单调区间范围的影响。
    \item \textbf{极值与最值}：求闭区间上的最值时，除端点外，还需比较各极值点处的函数值。
\end{enumerate}
