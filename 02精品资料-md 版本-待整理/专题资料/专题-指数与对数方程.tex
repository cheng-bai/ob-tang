    {\LARGE\bfseries 专题：指数与对数方程}
\end{center}

\section*{知识概要}
\addcontentsline{toc}{section}{知识概要}

指数方程与对数方程是含有未知数在指数位置或对数符号内的方程。解这类方程的基本思想是通过变形转化为代数方程求解。

\begin{definition}[指数方程]
    在指数里含有未知数的方程称为\textbf{指数方程}。
    
    基本类型：
    \begin{enumerate}
        \item $a^{f(x)} = a^{g(x)}$（$a > 0$，$a \neq 1$）$\Leftrightarrow f(x) = g(x)$
        \item $a^{f(x)} = b^{g(x)}$ 型：两边取对数转化为代数方程
        \item 可化为二次型的指数方程：设 $t = a^x$（$t > 0$）换元求解
    \end{enumerate}
\end{definition}

\begin{definition}[对数方程]
    在对数符号后面含有未知数的方程称为\textbf{对数方程}。
    
    基本类型：
    \begin{enumerate}
        \item $\log_a f(x) = \log_a g(x)$（$a > 0$，$a \neq 1$）$\Leftrightarrow \begin{cases} f(x) = g(x) \\ f(x) > 0 \end{cases}$
        \item $\log_a f(x) = b$ 型：转化为 $f(x) = a^b$
        \item 可化为二次型的对数方程：设 $t = \log_a x$ 换元求解
    \end{enumerate}
\end{definition}

\begin{property}[解指数方程的注意事项]
    \begin{enumerate}
        \item 利用指数函数的单调性：当 $a > 1$ 时，$a^{f(x)} = a^{g(x)} \Leftrightarrow f(x) = g(x)$；
        \item 注意验根：变形过程中可能产生增根，需要检验；
        \item 设元换元时注意新元的取值范围（如 $t = a^x > 0$）。
    \end{enumerate}
\end{property}

\begin{property}[解对数方程的注意事项]
    \begin{enumerate}
        \item 必须保证真数大于 $0$；
        \item 利用对数函数的单调性去掉对数符号；
        \item 解完后必须验根，舍去使真数 $\leq 0$ 或底数不合要求的根。
    \end{enumerate}
\end{property}

\section*{典型例题}
\addcontentsline{toc}{section}{典型例题}

\begin{example}[基本指数方程]
    解方程：$2^{x+1} = 16$
    
    \textbf{【来源】练习}
\end{example}

\textbf{解.} 将方程右边化为以 $2$ 为底的幂：$16 = 2^4$。

原方程化为：$2^{x+1} = 2^4$。

由于底数相同且 $2 > 0$，$2 \neq 1$，所以 $x + 1 = 4$，解得 $x = 3$。

经检验，$x = 3$ 是原方程的解。

\begin{example}[不同底数的指数方程]
    解方程：$3^x = 5^{x-1}$
    
    \textbf{【来源】练习}
\end{example}

\textbf{解.} 两边取常用对数（或自然对数）：

$x\lg 3 = (x-1)\lg 5$

展开：$x\lg 3 = x\lg 5 - \lg 5$

移项：$x(\lg 3 - \lg 5) = -\lg 5$

$x = \dfrac{-\lg 5}{\lg 3 - \lg 5} = \dfrac{\lg 5}{\lg 5 - \lg 3} = \dfrac{\lg 5}{\lg \frac{5}{3}}$

所以 $x = \log_{\frac{5}{3}} 5$。

\begin{example}[可化为二次型的指数方程]
    解方程：$4^x - 3 \cdot 2^x - 4 = 0$
    
    \textbf{【来源】练习}
\end{example}

\textbf{解.} 设 $t = 2^x$（$t > 0$），则 $4^x = (2^2)^x = (2^x)^2 = t^2$。

原方程化为：$t^2 - 3t - 4 = 0$。

因式分解：$(t-4)(t+1) = 0$，解得 $t = 4$ 或 $t = -1$。

因为 $t = 2^x > 0$，所以舍去 $t = -1$。

由 $2^x = 4 = 2^2$，得 $x = 2$。

经检验，$x = 2$ 是原方程的解。

\begin{example}[基本对数方程]
    解方程：$\log_2(x-1) + \log_2(x+1) = 3$
    
    \textbf{【来源】练习}
\end{example}

\textbf{解.} 首先确定定义域：$\begin{cases} x - 1 > 0 \\ x + 1 > 0 \end{cases}$，即 $x > 1$。

利用对数运算法则：$\log_2[(x-1)(x+1)] = 3$

即 $\log_2(x^2-1) = 3$

转化为指数式：$x^2 - 1 = 2^3 = 8$

$x^2 = 9$，解得 $x = 3$ 或 $x = -3$。

检验：$x = 3$ 满足 $x > 1$；$x = -3$ 不满足 $x > 1$，舍去。

所以原方程的解为 $x = 3$。

\begin{example}[可化为二次型的对数方程]
    解方程：$\log_3^2 x - 2\log_3 x - 3 = 0$
    
    \textbf{【来源】练习}
\end{example}

\textbf{解.} 设 $t = \log_3 x$，则原方程化为：$t^2 - 2t - 3 = 0$。

因式分解：$(t-3)(t+1) = 0$，解得 $t = 3$ 或 $t = -1$。

当 $t = 3$ 时，$\log_3 x = 3$，得 $x = 3^3 = 27$。

当 $t = -1$ 时，$\log_3 x = -1$，得 $x = 3^{-1} = \dfrac{1}{3}$。

检验：$x = 27 > 0$，$x = \dfrac{1}{3} > 0$，均满足定义域要求。

所以原方程的解为 $x = 27$ 或 $x = \dfrac{1}{3}$。

\begin{realexam}[2024 上海虹口二模]
    欧拉公式 $\mathrm{e}^{i\theta} = \cos\theta + i\sin\theta$ 把自然对数的底数 $\mathrm{e}$，虚数单位 $i$，三角函数 $\cos\theta$ 和 $\sin\theta$ 联系在一起，被誉为"数学的天桥"。若复数 $z$ 满足 $\left(z + \mathrm{e}^{i \cdot \frac{3\pi}{2}}\right) \cdot i = 2 - i$，则 $|z| =$（\quad）

    \text{A. } $\sqrt{2}$ \quad \text{B. } $2$ \quad \text{C. } $\sqrt{5}$ \quad \text{D. } $\sqrt{10}$
\end{realexam}

\textbf{解.} 首先计算 $\mathrm{e}^{i \cdot \frac{3\pi}{2}}$：

$\mathrm{e}^{i \cdot \frac{3\pi}{2}} = \cos\dfrac{3\pi}{2} + i\sin\dfrac{3\pi}{2} = 0 + i(-1) = -i$

设 $z = a + bi$（$a, b \in \mathbb{R}$），则：

$z + \mathrm{e}^{i \cdot \frac{3\pi}{2}} = a + bi - i = a + (b-1)i$

原方程化为：$[a + (b-1)i] \cdot i = 2 - i$

左边 $= ai + (b-1)i^2 = ai - (b-1) = -(b-1) + ai = (1-b) + ai$

所以 $\begin{cases} 1 - b = 2 \\ a = -1 \end{cases}$，解得 $\begin{cases} a = -1 \\ b = -1 \end{cases}$

因此 $z = -1 - i$，$|z| = \sqrt{(-1)^2 + (-1)^2} = \sqrt{2}$，选 A。

\section*{方法总结}
\addcontentsline{toc}{section}{方法总结}

\begin{enumerate}
    \item \textbf{同底化}：解指数方程时，尽量将不同底数化为同底数，利用单调性求解。
    \item \textbf{取对数法}：对于 $a^{f(x)} = b^{g(x)}$ 型方程，两边取对数是常用方法。
    \item \textbf{换元法}：对于可化为二次型的指对数方程，设元换元是有效策略，注意新元的取值范围。
    \item \textbf{定义域检验}：解对数方程必须首先确定定义域（真数 $> 0$），解完后必须验根。
    \item \textbf{数形结合}：对于复杂的指对数方程，可借助函数图像分析解的个数和范围。
    \item \textbf{指数与对数的联系}：指对数方程可以相互转化，$a^x = N \Leftrightarrow x = \log_a N$。
\end{enumerate}
