% ======================================
% 08-layout.tex - 页面布局与章节样式
% ======================================

% ========== 页面布局 ==========
\setlength{\headheight}{24pt}
\linespread{1.3}

% ========== 章节标题样式 ==========
\ctexset{
    chapter = {
        format = \centering,
        nameformat = \fontsize{32pt}{40pt}\selectfont\heavykaishu\color{LogicColor},
        titleformat = \fontsize{32pt}{40pt}\selectfont\heavykaishu\color{LogicColor},
        aftername = \hspace{1.5em},
        beforeskip = 8pt,
        afterskip = 40pt,
    },
    section = {
        format = \centering\Large\bfseries\color{MainText},
        beforeskip = 20pt,
        afterskip = 22pt,
    },
    subsection = {
        format = \large\bfseries\color{MainText},
        beforeskip = 16pt,
        afterskip = 12pt,
    }
}

% ========== 页眉页脚设置 ==========
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small \kaishu \LectureNotesTitle}
\fancyhead[R]{\small \kaishu \AuthorInfo}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.5pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{LogicColor}\leaders\hrule height \headrulewidth\hfill}}

\fancypagestyle{plain}{
    \fancyhf{}
    \fancyhead[L]{\small \kaishu \LectureNotesTitle}
    \fancyhead[R]{\small \kaishu \AuthorInfo}
    \fancyfoot[C]{\thepage}
    \renewcommand{\headrulewidth}{0.5pt}
    \renewcommand{\headrule}{\hbox to\headwidth{\color{LogicColor}\leaders\hrule height \headrulewidth\hfill}}
}

% ========== 公式编号 ==========
\numberwithin{equation}{chapter}

% ========== 超链接设置 ==========
\@ifundefined{hypersetup}{
    \hypersetup{
        colorlinks=true,
        linkcolor=LogicColor,
        citecolor=LogicColor,
        urlcolor=ExColor
    }
}{}
